   680→function renderSparkline() {
   681→  const data = state.sparkline;
   682→  if (data.length < 2) return;
   683→
   684→  const svg = document.getElementById('sparkline');
   685→  const w = 600, h = 60;
   686→  const min = Math.min(...data);
   687→  const max = Math.max(...data);
   688→  const range = max - min || 1;
   689→
   690→  const points = data.map((v, i) => {
   691→    const x = (i / (data.length - 1)) * w;
   692→    const y = h - ((v - min) / range) * (h - 4) - 2;
   693→    return `${x},${y}`;
   694→  }).join(' ');
   695→
   696→  const areaPoints = `0,${h} ${points} ${w},${h}`;
   697→  const isUp = data[data.length-1] >= data[0];
   698→  const color = isUp ? '#22c55e' : '#ef4444';
   699→
   700→  svg.setAttribute('viewBox', `0 0 ${w} ${h}`);
   701→  svg.innerHTML = `
   702→    <defs>
   703→      <linearGradient id="sparkGrad" x1="0" y1="0" x2="0" y2="1">
   704→        <stop offset="0%" stop-color="${color}" stop-opacity="0.3"/>
   705→        <stop offset="100%" stop-color="${color}" stop-opacity="0"/>
   706→      </linearGradient>
   707→    </defs>
   708→    <polygon points="${areaPoints}" fill="url(#sparkGrad)"/>
   709→    <polyline points="${points}" fill="none" stroke="${color}" stroke-width="1.5" stroke-linejoin="round"/>
   710→    <circle cx="${(data.length-1)/(data.length-1)*w}" cy="${h - ((data[data.length-1]-min)/range)*(h-4)-2}" r="3" fill="${color}"/>
   711→  `;
   712→}

   917→function renderKeyLevels() {
   918→  const { usd } = state.price;
   919→  const s = state.signal;
   920→  let levels = [];
   921→
   922→  // Derive key levels from available data
   923→  // MVRV = price / realized_price, so realized_price = price / MVRV
   924→  const rp = state.onchain.mvrv > 0 ? usd / state.onchain.mvrv : 0;
   925→  // LTH-RP approximation (typically 1.15x realized price)
   926→  const lthRp = rp * 1.15;
   927→
   928→  if (rp > 0) {
   929→    levels.push({ label: 'Realized Price (derived)', value: rp, dist: ((usd - rp) / rp) * 100 });
   930→  }
   931→  if (lthRp > 0) {
   932→    levels.push({ label: 'LTH Realized Price (est.)', value: lthRp, dist: ((usd - lthRp) / lthRp) * 100 });
   933→  }
   934→  levels.push({ label: 'ATH', value: CONFIG.ATH_USD, dist: ((usd - CONFIG.ATH_USD) / CONFIG.ATH_USD) * 100 });
   935→
   936→  // Compute THB/USD rate dynamically if possible
   937→  const thbUsdRate = (state.price.thb > 0 && state.price.usd > 0) ? state.price.usd / state.price.thb : 35;
   938→
   939→  // Add from bot state if available (match by label, never overwrite ATH)
   940→  if (state.bot && state.bot.last_indicators) {
   941→    const bi = state.bot.last_indicators;
   942→    if (bi.realized_price > 0) {
   943→      const rpUsd = bi.realized_price / thbUsdRate;
   944→      const idx = levels.findIndex(l => l.label.includes('Realized Price') && !l.label.includes('LTH'));
   945→      if (idx >= 0) { levels[idx] = { label: 'Realized Price', value: rpUsd, dist: ((usd - rpUsd) / rpUsd) * 100 }; }
   946→      else { levels.splice(levels.length - 1, 0, { label: 'Realized Price', value: rpUsd, dist: ((usd - rpUsd) / rpUsd) * 100 }); }
   947→    }
   948→    if (bi.lth_realized_price > 0) {
   949→      const lthUsd = bi.lth_realized_price / thbUsdRate;
   950→      const idx = levels.findIndex(l => l.label.includes('LTH'));
   951→      if (idx >= 0) { levels[idx] = { label: 'LTH Realized Price', value: lthUsd, dist: ((usd - lthUsd) / lthUsd) * 100 }; }
   952→      else { levels.splice(levels.length - 1, 0, { label: 'LTH Realized Price', value: lthUsd, dist: ((usd - lthUsd) / lthUsd) * 100 }); }
   953→    }
   954→  }
   955→
   956→  let html = '<div style="display:grid;grid-template-columns:repeat(auto-fit,minmax(200px,1fr));gap:0">';
   957→  levels.forEach(l => {
   958→    const dColor = l.dist >= 0 ? 'var(--green)' : 'var(--red)';
   959→    html += `<div class="level-row">`;
   960→    html += `  <span class="level-label">${l.label}</span>`;
   961→    html += `  <div class="level-right">`;
   962→    html += `    <span class="level-value">${fmtUSD(l.value)}</span>`;
   963→    html += `    <span class="level-dist" style="color:${dColor}">${l.dist >= 0 ? '+' : ''}${l.dist.toFixed(1)}%</span>`;
   964→    html += `  </div>`;
   965→    html += `</div>`;
   966→  });
   967→  html += '</div>';
   968→  html += `<div class="level-note">`;
   969→  html += `  Below Realized Price = historically profitable accumulation zone. `;
   970→  html += `  Above LTH-RP = long-term holders in profit (sell pressure zone). `;
   971→  html += `  Levels derived from MVRV: RP ≈ Price / MVRV.`;
   972→  html += `</div>`;
   973→
   974→  document.getElementById('levels-content').innerHTML = html;
   975→}
