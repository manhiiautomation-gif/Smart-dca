    61→def _cache_path(name):
    62→    """Return CSV cache file path for a given data source."""
    63→    return os.path.join(CACHE_DIR, f'{name}.csv')
    64→
    65→
    66→CACHE_MAX_AGE_HOURS = 168  # 7 days — historical data doesn't change
    67→
    68→
    69→def _load_cache(name):
    70→    """Load DataFrame from CSV cache if it exists and is recent (< 7 days)."""
    71→    path = _cache_path(name)
    72→    if os.path.exists(path):
    73→        age_hours = (time.time() - os.path.getmtime(path)) / 3600
    74→        if age_hours < CACHE_MAX_AGE_HOURS:
    75→            df = pd.read_csv(path, parse_dates=['date'])
    76→            df['date'] = df['date'].dt.date
    77→            print(f"[CACHE] Loaded {name} from disk ({len(df)} rows, {age_hours:.1f}h old)")
    78→            return df
    79→        else:
    80→            print(f"[CACHE] {name} cache expired ({age_hours:.1f}h old), re-fetching...")
    81→    return None
    82→

    84→def _save_cache(name, df):
    85→    """Save DataFrame to CSV cache."""
    86→    path = _cache_path(name)
    87→    df.to_csv(path, index=False)
    88→    print(f"[CACHE] Saved {name} to disk ({len(df)} rows)")

    91→def fetch_binance_btc_price(days=2000):
    92→    """
    93→    Fetch REAL BTC daily close prices from Binance Spot API (klines).
    94→    Uses pagination (2 x 1000) for ~2000 days of historical data.
    95→    Results are cached to CSV — subsequent runs use cache (no API call).
    96→    Returns DataFrame with ['date', 'price_usd'] or None on failure.
    97→    """
    98→    # Try cache first
    99→    cached = _load_cache('binance_btc_prices')
   100→    if cached is not None:
   101→        return cached
   102→
   103→    try:
   104→        print("[DATA] Fetching REAL BTC prices from Binance API...")
   105→        limit = 1000
   106→        url1 = (f"https://api.binance.com/api/v3/klines?symbol=BTCUSDT"
   107→               f"&interval=1d&limit={limit}")
   108→        resp1 = requests.get(url1, timeout=15)
   109→        if resp1.status_code != 200:
   110→            print(f"[DATA] Binance returned status {resp1.status_code}.")
   111→            return None
   112→        candles1 = resp1.json()
   113→        if not candles1:
   114→            return None
   115→
   116→        # Page 2: older 1000 days
   117→        first_time = candles1[0][0]
   118→        url2 = (f"https://api.binance.com/api/v3/klines?symbol=BTCUSDT"
   119→               f"&interval=1d&limit={limit}&endTime={first_time - 1}")
   120→        resp2 = requests.get(url2, timeout=15)
   121→        candles2 = resp2.json() if resp2.status_code == 200 else []
   122→
   123→        all_candles = candles2 + candles1
   124→        records = []
   125→        for c in all_candles:
   126→            dt = datetime.fromtimestamp(c[0] / 1000, tz=None).date()
   127→            close_price = float(c[4])
   128→            records.append({'date': dt, 'price_usd': close_price})
   129→
   130→        df = pd.DataFrame(records).drop_duplicates(subset='date').sort_values('date').reset_index(drop=True)
   131→        print(f"[DATA] Binance returned {len(df)} days of REAL price data.")
   132→        print(f"        Period: {df['date'].iloc[0]} to {df['date'].iloc[-1]}")
   133→
   134→        # Save to cache
   135→        _save_cache('binance_btc_prices', df)

   142→def fetch_coinmetrics_mvrv(start_date, end_date):
   143→    """
   144→    Fetch REAL MVRV from CoinMetrics Community API (FREE, no API key needed).
   145→    Metric: CapMVRVCur (Market Value to Realized Value Ratio).
   146→    High rate limit — no issues in practice.
   147→    Cached to CSV — subsequent runs skip the API entirely.
   148→    Returns DataFrame with ['date', 'mvrv'] or None on failure.
   149→    """
   150→    cache_key = 'coinmetrics_mvrv'
   151→    cached = _load_cache(cache_key)
   152→    if cached is not None:
   153→        return cached
   154→
   155→    try:
   156→        print(f"[DATA] Fetching REAL MVRV from CoinMetrics Community API...")
   157→        all_records = []
   158→        # CoinMetrics limits ~100 rows per request — use 90-day chunks
   159→        current_start = start_date.strftime('%Y-%m-%dT00:00:00Z')
   160→        end_str = end_date.strftime('%Y-%m-%dT23:59:59Z')
   161→        
   162→        while current_start < end_str:
   163→            chunk_end_dt = (pd.to_datetime(current_start) + timedelta(days=90)).strftime('%Y-%m-%dT00:00:00Z')
   164→            if chunk_end_dt > end_str:
   165→                chunk_end_dt = end_str
   166→            
   167→            url = "https://community-api.coinmetrics.io/v4/timeseries/asset-metrics"
   168→            params = {
   169→                'assets': 'btc',
   170→                'metrics': 'CapMVRVCur',
   171→                'frequency': '1d',
   172→                'start_time': current_start,
   173→                'end_time': chunk_end_dt,
   174→            }
   175→            resp = requests.get(url, params=params, timeout=30)
   176→            if resp.status_code != 200:
   177→                print(f"[DATA] CoinMetrics HTTP {resp.status_code}: {resp.text[:200]}")
   178→                if not all_records:
   179→                    return None
   180→                break
   181→            
   182→            data = resp.json().get('data', [])
   183→            for item in data:
   184→                dt = pd.to_datetime(item['time']).date()
   185→                val = float(item['CapMVRVCur'])
   186→                if val > 0:  # skip invalid MVRV
   187→                    all_records.append({'date': dt, 'mvrv': val})
   188→            
   189→            print(f"[DATA]   {current_start[:10]} to {chunk_end_dt[:10]}: {len(data)} rows")
   190→            current_start = chunk_end_dt
   191→            time.sleep(0.2)  # CoinMetrics has high rate limits
   192→        
   193→        if not all_records:
   194→            return None
   195→        
   196→        df = pd.DataFrame(all_records).drop_duplicates(subset='date').sort_values('date').reset_index(drop=True)
   197→        print(f"[DATA] CoinMetrics MVRV: {len(df)} records ({df['date'].iloc[0]} to {df['date'].iloc[-1]})")
   198→        print(f"        MVRV range: {df['mvrv'].min():.3f} - {df['mvrv'].max():.3f}")
   199→        _save_cache(cache_key, df)
   200→        return df
   201→    except Exception as e:
   202→        print(f"[DATA] CoinMetrics MVRV fetch failed: {e}")
   203→        return None

   206→def fetch_bgeometrics_metric(metric_name, token='7NqNRwWhyc'):
   207→    """
   208→    Fetch REAL on-chain metric from BGeometrics API (FALLBACK only).
   209→    Used for SOPR which CoinMetrics doesn't offer for free.
   210→    """
   211→    cache_key = f'bgeometrics_{metric_name}'
   212→    cached = _load_cache(cache_key)
   213→    if cached is not None:
   214→        return cached
   215→
   216→    url = f"https://api.bgeometrics.com/v1/{metric_name}?token={token}"
   217→    for attempt in range(3):
   218→        try:
   219→            if attempt > 0:
   220→                wait = 10 * attempt
   221→                print(f"[DATA] BGeometrics {metric_name}: retry in {wait}s (attempt {attempt+1}/3)")
   222→                time.sleep(wait)
   223→            resp = requests.get(url, headers={'User-Agent': 'Mozilla/5.0'}, timeout=15)
   224→            if resp.status_code == 429:
   225→                print(f"[DATA] BGeometrics {metric_name}: HTTP 429 rate-limit (attempt {attempt+1}/3)")
   226→                continue
   227→            if resp.status_code != 200:
   228→                return None
   229→            data = resp.json()
   230→            if not isinstance(data, list) or len(data) == 0:
   231→                return None
   232→            records = []
   233→            for item in data:
   234→                date_str = item.get('d')
   235→                metric_val = item.get(metric_name)
   236→                if date_str and metric_val is not None:
   237→                    dt = pd.to_datetime(date_str).date()
   238→                    records.append({'date': dt, metric_name: float(metric_val)})
   239→            if not records:
   240→                return None
   241→            df = pd.DataFrame(records).drop_duplicates(subset='date').sort_values('date').reset_index(drop=True)
   242→            print(f"[DATA] BGeometrics {metric_name}: {len(df)} records "
   243→                  f"({df['date'].iloc[0]} to {df['date'].iloc[-1]})")
   244→            _save_cache(cache_key, df)
   245→            return df
   246→        except Exception as e:
   247→            if attempt == 2:
   248→                return None
   249→    return None

   252→def generate_mock_btc_prices(start_date, end_date, seed=42):
   253→    """
   254→    Generate realistic mock BTC daily prices (LAST RESORT only if API fails).
   255→    Simulates a full BTC cycle: Bull -> Peak -> Bear -> Recovery -> New Bull.
   256→    """
   257→    np.random.seed(seed)
   258→    dates = pd.date_range(start=start_date, end=end_date, freq='D')
   259→    n = len(dates)
   260→    waypoints = [
   261→        (0,     np.log(30000)), (200,   np.log(55000)), (350,   np.log(69000)),
   262→        (420,   np.log(45000)), (600,   np.log(22000)), (800,   np.log(16000)),
   263→        (1000,  np.log(25000)), (1200,  np.log(42000)), (1400,  np.log(65000)),
   264→        (1550,  np.log(95000)), (1700,  np.log(110000)), (n - 1, np.log(98000)),
   265→    ]
   266→    wp_days = [w[0] for w in waypoints]
   267→    wp_logs = [w[1] for w in waypoints]
   268→    all_days = np.arange(n)
   269→    log_prices = np.interp(all_days, wp_days, wp_logs)
   270→    sigma = 0.50
   271→    dt_step = 1 / 365.25
   272→    noise = np.random.standard_normal(n) * sigma * np.sqrt(dt_step)
   273→    log_prices += noise
   274→    log_prices = pd.Series(log_prices).ewm(span=3, adjust=False).mean().values
   275→    prices = np.exp(log_prices)
   276→    prices = np.clip(prices, 5000, 200000)
   277→    df = pd.DataFrame({'date': dates.date, 'price_usd': prices})
   278→    print(f"[DATA] MOCK FALLBACK: {n} days ({start_date} to {end_date}).")

   283→def generate_mock_onchain_metrics(price_df, seed=123):
   284→    """Generate mock on-chain metrics (LAST RESORT only)."""
   285→    np.random.seed(seed)
   286→    n = len(price_df)
   287→    dates = price_df['date'].values
   288→    prices = price_df['price_usd'].values
   289→    sma_365 = pd.Series(prices).rolling(365, min_periods=1).mean().values
   290→    price_ratio = prices / np.maximum(sma_365, 1)
   291→    cycle_phase = np.linspace(0, 3 * np.pi, n)
   292→    mvrv_base = np.power(price_ratio, 1.8)
   293→    mvrv_cycle = 1.5 * np.sin(cycle_phase)
   294→    mvrv_noise = np.random.normal(0, 0.08, n)
   295→    mvrv = mvrv_base + mvrv_cycle + mvrv_noise
   296→    mvrv = pd.Series(np.clip(mvrv, 0.4, 7.0)).ewm(span=14).mean().values
   297→    nupl_base = (prices - sma_365) / np.maximum(prices, 1)
   298→    nupl = nupl_base * 2.5 + np.random.normal(0, 0.08, n)
   299→    nupl = pd.Series(np.clip(nupl, -0.5, 0.8)).ewm(span=7).mean().values
   300→    sopr_noise = np.random.normal(0, 0.05, n)
   301→    sopr_trend = 1.0 + 0.3 * (price_ratio - 1.0) + np.sin(cycle_phase * 0.7) * 0.1
   302→    sopr = pd.Series(np.clip(sopr_trend + sopr_noise, 0.5, 2.5)).ewm(span=7).mean().values
   303→    df = pd.DataFrame({'date': dates, 'mvrv': mvrv, 'nupl': nupl, 'sopr': sopr})
   304→    print(f"[DATA] MOCK on-chain metrics: {n} days.")
   305→    return df

   308→def compute_technical_indicators(df):
   309→    """Add EMA 20, RSI 14, SMA 365, MACD(12,26,9), and SMA 200 to the DataFrame."""
   310→    prices = df['price_usd']
   311→    df['ema_20'] = prices.ewm(span=20, adjust=False).mean()
   312→    delta = prices.diff()
   313→    gain = delta.where(delta > 0, 0.0).ewm(span=14, adjust=False).mean()
   314→    loss = (-delta.where(delta < 0, 0.0)).ewm(span=14, adjust=False).mean()
   315→    rs = gain / loss.replace(0, np.nan)
   316→    df['rsi_14'] = 100 - (100 / (1 + rs))
   317→    df['rsi_14'] = df['rsi_14'].fillna(50)
   318→    df['sma_365'] = prices.rolling(365, min_periods=1).mean()
   319→    df['sma_200'] = prices.rolling(200, min_periods=50).mean()
   320→    # MACD (12, 26, 9)
   321→    ema12 = prices.ewm(span=12, adjust=False).mean()
   322→    ema26 = prices.ewm(span=26, adjust=False).mean()
   323→    df['macd_line'] = ema12 - ema26
   324→    df['macd_signal'] = df['macd_line'].ewm(span=9, adjust=False).mean()
   325→    df['macd_hist'] = df['macd_line'] - df['macd_signal']
   326→    # Realized Price = Price / MVRV (exact derivation)
   327→    df['realized_price'] = df['price_usd'] / df['mvrv'].clip(lower=0.01)
   328→    # LTH Realized Price proxy: 180-day SMA of Realized Price
   329→    # Rationale: LTH cost basis moves slowly; SMA180 of RP approximates this
   330→    # NOTE: Correlation with MVRV is ~0.993 — used as CONFIRMATION, not primary signal
   331→    df['lth_realized_price'] = df['realized_price'].rolling(180, min_periods=60).mean()
   332→    df['price_to_lth_rp'] = df['price_usd'] / df['lth_realized_price'].clip(lower=1)
   333→    return df

   336→def build_master_dataframe(years=5):
   337→    """
   338→    Main data pipeline: fetch REAL data (with CSV cache), merge into master DF.
   339→    Falls back to mock data ONLY if both API and cache fail.
   340→    Falls back to proxy calculations for dates before BGeometrics data starts.
   341→    """
   342→    used_cache = {'price': False, 'mvrv': False, 'nupl': False, 'sopr': False}
   343→
   344→    # --- Step 1: REAL BTC Price from Binance (or cache) ---
   345→    price_df = fetch_binance_btc_price(days=2000)
   346→    if price_df is None or len(price_df) < 365:
   347→        # Check if there's an old cache we can still use
   348→        old_cache = pd.read_csv(_cache_path('binance_btc_prices'), parse_dates=['date'])
   349→        if len(old_cache) >= 365:
   350→            old_cache['date'] = old_cache['date'].dt.date
   351→            price_df = old_cache
   352→            print("[DATA] Using expired cache as fallback for prices.")
   353→        else:
   354→            print("[DATA] Binance FAILED. Using mock prices as LAST RESORT.")
   355→            end_date = datetime.now().date()
   356→            start_date = end_date - timedelta(days=int(years * 365.25))
   357→            price_df = generate_mock_btc_prices(start_date, end_date)
   358→    else:
   359→        used_cache['price'] = os.path.exists(_cache_path('binance_btc_prices'))
   360→
   361→    # --- Step 2: REAL On-Chain Metrics ---
   362→    # PRIMARY:   CoinMetrics Community API (CapMVRVCur) — FREE, high rate limit
   363→    # DERIVED:   NUPL = 1 - (1/MVRV) — mathematically identical to on-chain NUPL
   364→    # FALLBACK:  BGeometrics SOPR (rate-limited) → Proxy (Price/EMA30)
   365→    data_start = price_df['date'].iloc[0]
   366→    data_end = price_df['date'].iloc[-1]
   367→
   368→    print(f"\n[DATA] === ON-CHAIN DATA PIPELINE ===")
   369→    print(f"        Primary source: CoinMetrics Community API (CapMVRVCur)")
   370→    print(f"        NUPL: Derived from MVRV (NUPL = 1 - 1/MVRV)")
   371→    print(f"        SOPR: BGeometrics fallback -> Proxy (Price/EMA30)")
   372→
   373→    # Fetch REAL MVRV from CoinMetrics
   374→    mvrv_df = fetch_coinmetrics_mvrv(data_start, data_end)
   375→    nupl_df = None  # Will be derived from MVRV
   376→
   377→    # Try BGeometrics SOPR (non-critical, proxy is acceptable)
   378→    time.sleep(2)
   379→    sopr_df = fetch_bgeometrics_metric('sopr')
   380→
   381→    # Merge on-chain data
   382→    onchain_df = price_df[['date']].copy()
   383→    onchain_df['mvrv'] = np.nan
   384→    onchain_df['nupl'] = np.nan
   385→    onchain_df['sopr'] = np.nan
   386→    if mvrv_df is not None:
   387→        onchain_df = onchain_df.drop(columns=['mvrv']).merge(mvrv_df, on='date', how='left')
   388→    if sopr_df is not None:
   389→        onchain_df = onchain_df.drop(columns=['sopr']).merge(sopr_df, on='date', how='left')
   390→
   391→    master = price_df.merge(onchain_df, on='date', how='left')
   392→    master = master.sort_values('date').reset_index(drop=True)
   393→
   394→    # Forward-fill missing on-chain (up to 2 days)
   395→    for col in ['mvrv', 'nupl', 'sopr']:
   396→        master[col] = master[col].ffill(limit=2)
   397→
   398→    # --- Step 3: Proxy fallback ---
   399→    master['sma_365'] = master['price_usd'].rolling(365, min_periods=1).mean()
   400→    # MVRV proxy (only if CoinMetrics failed)
   401→    master['mvrv_proxy'] = master['price_usd'] / master['sma_365']
   402→    master['mvrv'] = master['mvrv'].fillna(master['mvrv_proxy'])
   403→
   404→    # NUPL: DERIVE from real MVRV where available, proxy otherwise
   405→    # Real: NUPL = 1 - (1/MVRV)   [on-chain identity]
   406→    # Proxy: NUPL = (Price - SMA365) / Price
   407→    master['nupl_real'] = 1.0 - (1.0 / master['mvrv'])
   408→    nupl_proxy = (master['price_usd'] - master['sma_365']) / master['price_usd']
   409→    # Use derived NUPL where MVRV is real (not proxy), else proxy
   410→    master['nupl'] = master['nupl'].fillna(master['nupl_real'])
   411→    master['nupl'] = master['nupl'].fillna(nupl_proxy)
   412→    master.drop(columns=['nupl_real'], inplace=True)
   413→
   414→    # SOPR proxy: Price/EMA30
   415→    ema30 = master['price_usd'].ewm(span=30, adjust=False).mean()
   416→    sopr_proxy = master['price_usd'] / ema30
   417→    master['sopr'] = master['sopr'].fillna(sopr_proxy)
   418→
   419→    # Report data coverage
   420→    real_mvrv_count = len(mvrv_df) if mvrv_df is not None else 0
   421→    real_sopr_count = len(sopr_df) if sopr_df is not None else 0
   422→    # NUPL is derived from MVRV, so same count
   423→    real_nupl_count = real_mvrv_count
   424→
   425→    print(f"\n[DATA] === DATA SOURCE SUMMARY ===")
   426→    print(f"        Price:  Binance REAL ({len(master)} days)")
   427→    print(f"        MVRV:   CoinMetrics REAL ({real_mvrv_count}d) + Proxy ({max(0, len(master)-real_mvrv_count)}d)")
   428→    print(f"        NUPL:   Derived from MVRV ({real_nupl_count}d) + Proxy ({max(0, len(master)-real_nupl_count)}d)")
   429→    print(f"        SOPR:   BGeometrics REAL ({real_sopr_count}d) + Proxy ({max(0, len(master)-real_sopr_count)}d)")
   430→    print(f"        Cache dir: {CACHE_DIR}/")
   431→    print(f"        Cache TTL : {CACHE_MAX_AGE_HOURS}h (7 days)")
   432→    print(f"        To force re-fetch: rm {CACHE_DIR}/*.csv")
   433→
   434→    if real_mvrv_count > 0:
   435→        real_mvrv_range = master[master['mvrv'].notna()]['mvrv']
   436→        # Get the real MVRV range (not proxy)

   463→def apply_buy_fee(thb_amount):
   464→    return thb_amount * (1 - BUY_FEE_PCT)
   465→
   466→def apply_sell_fee(thb_amount):
   467→    return thb_amount * (1 - SELL_FEE_PCT)
   468→
   469→def backtest_strategy(df, strategy_func, strategy_name):
   470→    """
   471→    Generic backtest runner. Calls strategy_func for each day.
   472→    Tracks net_capital (new user money only, excluding reserve recycling)
   473→    for accurate ROI of reserve-using strategies.
   474→    """
   475→    btc = 0.0
   476→    cash_reserve = 0.0
   477→    total_invested = 0.0
   478→    adjusted_invested = 0.0  # Reduces proportionally on sells (smooth avg cost)
   479→    net_capital = 0.0       # Only new user capital (excludes reserve recycling)
   480→    total_sell_proceeds = 0.0
   481→    total_reserve_injected = 0.0
   482→    cooldown = 0
   483→    short_cooldown = 0
   484→    peak_value = 0.0
   485→    max_drawdown = 0.0
   486→    daily_log = []
   487→
   488→    for idx, row in df.iterrows():
   489→        price_thb = row['price_thb']
   490→        if cooldown > 0:
   491→            cooldown -= 1
   492→        if short_cooldown > 0:
   493→            short_cooldown -= 1
   494→
   495→        state = {
   496→            'btc': btc, 'cash_reserve': cash_reserve,
   497→            'total_invested': total_invested, 'cooldown': cooldown,
   498→            'short_cooldown': short_cooldown,
   499→            'row': row, 'idx': idx
   500→        }
   501→        action = strategy_func(state)
   502→
   503→        buy_thb = action.get('buy_thb', 0)
   504→        sell_btc_pct = action.get('sell_btc_pct', 0)
   505→        sell_thb = action.get('sell_thb', 0)
   506→        to_reserve = action.get('to_reserve', 0)
   507→        sell_score = action.get('sell_score', 0)
   508→        reserve_injection = action.get('reserve_injection', 0)
   509→
   510→        actual_buy = apply_buy_fee(buy_thb)
   511→        btc_bought = actual_buy / price_thb if price_thb > 0 else 0
   512→        btc_before_sell = btc + btc_bought  # BTC after buy, before sell
   513→        btc += btc_bought
   514→        total_invested += buy_thb
   515→        adjusted_invested += buy_thb
   516→        # Net capital: total buy minus what came from reserve
   517→        net_capital += buy_thb - reserve_injection
   518→        total_reserve_injected += reserve_injection
   519→
   520→        if sell_btc_pct > 0 and btc > 0:
   521→            btc_to_sell = btc * (sell_btc_pct / 100.0)
   522→            sell_proceeds = apply_sell_fee(btc_to_sell * price_thb)
   523→            btc -= btc_to_sell
   524→            cash_reserve += sell_proceeds
   525→            cooldown = action.get('new_cooldown', cooldown)
   526→            # Adjust invested: remove proportional cost basis of sold BTC
   527→            sell_frac = btc_to_sell / btc_before_sell if btc_before_sell > 0 else 0
   528→            adjusted_invested *= (1.0 - sell_frac)
   529→
   530→        # THB-based selling (Style Beta v3 / Omega)
   531→        if sell_thb > 0 and btc > 0 and price_thb > 0:
   532→            btc_to_sell = sell_thb / price_thb
   533→            if btc_to_sell > btc:
   534→                btc_to_sell = btc
   535→            sell_proceeds = apply_sell_fee(btc_to_sell * price_thb)
   536→            btc -= btc_to_sell
   537→            cash_reserve += sell_proceeds
   538→            total_sell_proceeds += sell_proceeds
   539→            cooldown = action.get('new_cooldown', cooldown)
   540→            # Adjust invested: remove proportional cost basis of sold BTC
   541→            sell_frac = btc_to_sell / btc_before_sell if btc_before_sell > 0 else 0
   542→            adjusted_invested *= (1.0 - sell_frac)
   543→
   544→        cash_reserve += to_reserve
   545→        cash_reserve = max(cash_reserve, 0.0)
   546→
   547→        portfolio_value = btc * price_thb + cash_reserve
   548→        avg_cost = total_invested / btc if btc > 0 else 0
   549→        adjusted_avg_cost = adjusted_invested / btc if btc > 0 else 0
   550→
   551→        if portfolio_value > peak_value:
   552→            peak_value = portfolio_value
   553→        if peak_value > 0:
   554→            drawdown = (peak_value - portfolio_value) / peak_value
   555→            if drawdown > max_drawdown:
   556→                max_drawdown = drawdown
   557→
   558→        daily_log.append({
   559→            'date': row['date'], 'price_thb': price_thb,
   560→            'btc': btc, 'cash_reserve': cash_reserve,
   561→            'total_invested': total_invested,
   562→            'portfolio_value': portfolio_value,
   563→            'avg_cost': adjusted_avg_cost,  # Use adjusted for chart (no sell spikes)
   564→            'max_drawdown_so_far': max_drawdown,
   565→        })
   566→
   567→    final_price = df.iloc[-1]['price_thb']
   568→    final_value = btc * final_price + cash_reserve
   569→    final_avg_cost = adjusted_invested / btc if btc > 0 else 0
   570→    # ROI based on total invested (all THB through buy side)
   571→    roi_pct = ((final_value - total_invested) / total_invested * 100) if total_invested > 0 else 0
   572→    net_profit = final_value - total_invested
   573→    # True ROI: only new user capital (excludes reserve recycling)
   574→    true_roi_pct = ((final_value - net_capital) / net_capital * 100) if net_capital > 0 else 0
   575→    true_net_profit = final_value - net_capital
   576→
   577→    results = {
   578→        'strategy': strategy_name,
   579→        'total_invested': total_invested,
   580→        'net_capital': net_capital,
   581→        'total_btc': btc,
   582→        'avg_cost_thb': final_avg_cost,
   583→        'avg_cost_usd': final_avg_cost / USD_THB_RATE,
   584→        'final_value': final_value,
   585→        'cash_reserve': cash_reserve,
   586→        'roi_pct': roi_pct,
   587→        'true_roi_pct': true_roi_pct,
   588→        'net_profit': net_profit,
   589→        'true_net_profit': true_net_profit,
   590→        'total_sell_proceeds': total_sell_proceeds,
   591→        'total_reserve_injected': total_reserve_injected,
   592→        'max_drawdown_pct': max_drawdown * 100,
   593→    }
   594→    return results, pd.DataFrame(daily_log)
