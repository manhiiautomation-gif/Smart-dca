   663→   663→function renderSparkline() {
   664→   664→  const data = state.sparkline;
   665→   665→  if (data.length < 2) return;
   666→   666→
   667→   667→  const svg = document.getElementById('sparkline');
   668→   668→  const w = 600, h = 60;
   669→   669→  const min = Math.min(...data);
   670→   670→  const max = Math.max(...data);
   671→   671→  const range = max - min || 1;
   672→   672→
   673→   673→  const points = data.map((v, i) => {
   674→   674→    const x = (i / (data.length - 1)) * w;
   675→   675→    const y = h - ((v - min) / range) * (h - 4) - 2;
   676→   676→    return `${x},${y}`;
   677→   677→  }).join(' ');
   678→   678→
   679→   679→  const areaPoints = `0,${h} ${points} ${w},${h}`;
   680→   680→  const isUp = data[data.length-1] >= data[0];
   681→   681→  const color = isUp ? '#22c55e' : '#ef4444';
   682→   682→
   683→   683→  svg.setAttribute('viewBox', `0 0 ${w} ${h}`);
   684→   684→  svg.innerHTML = `
   685→   685→    <defs>
   686→   686→      <linearGradient id="sparkGrad" x1="0" y1="0" x2="0" y2="1">
   687→   687→        <stop offset="0%" stop-color="${color}" stop-opacity="0.3"/>
   688→   688→        <stop offset="100%" stop-color="${color}" stop-opacity="0"/>
   689→   689→      </linearGradient>
   690→   690→    </defs>
   691→   691→    <polygon points="${areaPoints}" fill="url(#sparkGrad)"/>
   692→   692→    <polyline points="${points}" fill="none" stroke="${color}" stroke-width="1.5" stroke-linejoin="round"/>
   693→   693→    <circle cx="${(data.length-1)/(data.length-1)*w}" cy="${h - ((data[data.length-1]-min)/range)*(h-4)-2}" r="3" fill="${color}"/>
   694→   694→  `;
   695→   695→}

   865→   865→function renderKeyLevels() {
   866→   866→  const { usd } = state.price;

   868→   868→  let levels = [];

   873→   873→  // LTH-RP approximation (typically 1.15x realized price)

   877→   877→    levels.push({ label: 'Realized Price (derived)', value: rp, dist: ((usd - rp) / rp) * 100 });

   879→   879→  if (lthRp > 0) {
   880→   880→    levels.push({ label: 'LTH Realized Price (est.)', value: lthRp, dist: ((usd - lthRp) / lthRp) * 100 });
   881→   881→  }
