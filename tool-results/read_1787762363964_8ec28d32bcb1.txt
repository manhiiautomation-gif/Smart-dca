   676→function renderSparkline() {
   677→  const data = state.sparkline;
   678→  if (data.length < 2) return;
   679→
   680→  const svg = document.getElementById('sparkline');
   681→  const w = 600, h = 60;
   682→  const min = Math.min(...data);
   683→  const max = Math.max(...data);
   684→  const range = max - min || 1;
   685→
   686→  const points = data.map((v, i) => {
   687→    const x = (i / (data.length - 1)) * w;
   688→    const y = h - ((v - min) / range) * (h - 4) - 2;
   689→    return `${x},${y}`;
   690→  }).join(' ');
   691→
   692→  const areaPoints = `0,${h} ${points} ${w},${h}`;
   693→  const isUp = data[data.length-1] >= data[0];
   694→  const color = isUp ? '#22c55e' : '#ef4444';
   695→
   696→  svg.setAttribute('viewBox', `0 0 ${w} ${h}`);
   697→  svg.innerHTML = `
   698→    <defs>
   699→      <linearGradient id="sparkGrad" x1="0" y1="0" x2="0" y2="1">
   700→        <stop offset="0%" stop-color="${color}" stop-opacity="0.3"/>
   701→        <stop offset="100%" stop-color="${color}" stop-opacity="0"/>
   702→      </linearGradient>
   703→    </defs>
   704→    <polygon points="${areaPoints}" fill="url(#sparkGrad)"/>
   705→    <polyline points="${points}" fill="none" stroke="${color}" stroke-width="1.5" stroke-linejoin="round"/>
   706→    <circle cx="${(data.length-1)/(data.length-1)*w}" cy="${h - ((data[data.length-1]-min)/range)*(h-4)-2}" r="3" fill="${color}"/>
   707→  `;
   708→}

   913→function renderKeyLevels() {
   914→  const { usd } = state.price;
   915→  const s = state.signal;
   916→  let levels = [];
   917→
   918→  // Derive key levels from available data
   919→  // MVRV = price / realized_price, so realized_price = price / MVRV
   920→  const rp = state.onchain.mvrv > 0 ? usd / state.onchain.mvrv : 0;
   921→  // LTH-RP approximation (typically 1.15x realized price)
   922→  const lthRp = rp * 1.15;
   923→
   924→  if (rp > 0) {
   925→    levels.push({ label: 'Realized Price (derived)', value: rp, dist: ((usd - rp) / rp) * 100 });
   926→  }
   927→  if (lthRp > 0) {
   928→    levels.push({ label: 'LTH Realized Price (est.)', value: lthRp, dist: ((usd - lthRp) / lthRp) * 100 });
   929→  }
   930→  levels.push({ label: 'ATH', value: CONFIG.ATH_USD, dist: ((usd - CONFIG.ATH_USD) / CONFIG.ATH_USD) * 100 });
   931→
   932→  // Add from bot state if available
   933→  if (state.bot && state.bot.last_indicators) {
   934→    const bi = state.bot.last_indicators;
   935→    if (bi.realized_price > 0) {
   936→      levels[0] = { label: 'Realized Price', value: bi.realized_price / 35, dist: ((usd - bi.realized_price/35) / (bi.realized_price/35)) * 100 };
   937→    }
   938→    if (bi.lth_realized_price > 0) {
   939→      levels[1] = { label: 'LTH Realized Price', value: bi.lth_realized_price / 35, dist: ((usd - bi.lth_realized_price/35) / (bi.lth_realized_price/35)) * 100 };
   940→    }
   941→  }
   942→
   943→  let html = '<div style="display:grid;grid-template-columns:repeat(auto-fit,minmax(200px,1fr));gap:0">';
   944→  levels.forEach(l => {
   945→    const dColor = l.dist >= 0 ? 'var(--green)' : 'var(--red)';
   946→    html += `<div class="level-row">`;
   947→    html += `  <span class="level-label">${l.label}</span>`;
   948→    html += `  <div class="level-right">`;
   949→    html += `    <span class="level-value">${fmtUSD(l.value)}</span>`;
   950→    html += `    <span class="level-dist" style="color:${dColor}">${l.dist >= 0 ? '+' : ''}${l.dist.toFixed(1)}%</span>`;
   951→    html += `  </div>`;
   952→    html += `</div>`;
   953→  });
   954→  html += '</div>';
   955→  html += `<div class="level-note">`;
   956→  html += `  Below Realized Price = historically profitable accumulation zone. `;
   957→  html += `  Above LTH-RP = long-term holders in profit (sell pressure zone). `;
   958→  html += `  Levels derived from MVRV: RP ≈ Price / MVRV.`;
   959→  html += `</div>`;
   960→
   961→  document.getElementById('levels-content').innerHTML = html;
   962→}
