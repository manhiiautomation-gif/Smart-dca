   662→function renderSparkline() {
   663→  const data = state.sparkline;
   664→  if (data.length < 2) return;
   665→
   666→  const svg = document.getElementById('sparkline');
   667→  const w = 600, h = 60;
   668→  const min = Math.min(...data);
   669→  const max = Math.max(...data);
   670→  const range = max - min || 1;
   671→
   672→  const points = data.map((v, i) => {
   673→    const x = (i / (data.length - 1)) * w;
   674→    const y = h - ((v - min) / range) * (h - 4) - 2;
   675→    return `${x},${y}`;
   676→  }).join(' ');
   677→
   678→  const areaPoints = `0,${h} ${points} ${w},${h}`;
   679→  const isUp = data[data.length-1] >= data[0];
   680→  const color = isUp ? '#22c55e' : '#ef4444';
   681→
   682→  svg.setAttribute('viewBox', `0 0 ${w} ${h}`);
   683→  svg.innerHTML = `
   684→    <defs>
   685→      <linearGradient id="sparkGrad" x1="0" y1="0" x2="0" y2="1">
   686→        <stop offset="0%" stop-color="${color}" stop-opacity="0.3"/>
   687→        <stop offset="100%" stop-color="${color}" stop-opacity="0"/>
   688→      </linearGradient>
   689→    </defs>
   690→    <polygon points="${areaPoints}" fill="url(#sparkGrad)"/>
   691→    <polyline points="${points}" fill="none" stroke="${color}" stroke-width="1.5" stroke-linejoin="round"/>
   692→    <circle cx="${(data.length-1)/(data.length-1)*w}" cy="${h - ((data[data.length-1]-min)/range)*(h-4)-2}" r="3" fill="${color}"/>
   693→  `;
   694→}

   853→function renderKeyLevels() {
   854→  const { usd } = state.price;
   855→  const s = state.signal;
   856→  let levels = [];
   857→
   858→  // Derive key levels from available data
   859→  // MVRV = price / realized_price, so realized_price = price / MVRV
   860→  const rp = state.onchain.mvrv > 0 ? usd / state.onchain.mvrv : 0;
   861→  // LTH-RP approximation (typically 1.15x realized price)
   862→  const lthRp = rp * 1.15;
   863→
   864→  if (rp > 0) {
   865→    levels.push({ label: 'Realized Price (derived)', value: rp, dist: ((usd - rp) / rp) * 100 });
   866→  }
   867→  if (lthRp > 0) {

   992→      parts.push(`Signal Score: ${sc.toFixed(1)}/100 — Strong buy signal from multi-indicator analysis.`);

  1003→    }
