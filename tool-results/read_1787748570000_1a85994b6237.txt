   110→// ─── Mini Chart Component (SVG sparkline) ───
   111→
   112→function Sparkline({
   113→  data,
   114→  color = "#f59e0b",
   115→  height = 40,
   116→}: {
   117→  data: number[];
   118→  color?: string;
   119→  height?: number;
   120→}) {
   121→  if (data.length < 2) return null;
   122→  const min = Math.min(...data);
   123→  const max = Math.max(...data);
   124→  const range = max - min || 1;
   125→  const w = 100;
   126→  const points = data
   127→    .map((v, i) => {
   128→      const x = (i / (data.length - 1)) * w;
   129→      const y = height - ((v - min) / range) * (height - 4) - 2;
   130→      return `${x},${y}`;
   131→    })
   132→    .join(" ");
   133→  return (
   134→    <svg
   135→      viewBox={`0 0 ${w} ${height}`}
   136→      className="w-full"
   137→      style={{ height }}
   138→      preserveAspectRatio="none"
   139→    >
   140→      <polyline
   141→        points={points}
   142→        fill="none"
   143→        stroke={color}
   144→        strokeWidth="1.5"
   145→        strokeLinejoin="round"
   146→      />
   147→    </svg>
   148→  );
   149→}
   150→
   151→// ─── Format helpers ───
   152→
   153→function fmt(n: number, decimals = 2): string {
   154→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(decimals)}M`;
   155→  if (n >= 1_000) return `${(n / 1_000).toFixed(decimals)}K`;
   156→  return n.toFixed(decimals);
   157→}
   158→
   159→function fmtCurrency(n: number, currency: string): string {
   160→  return `${n.toLocaleString(undefined, { maximumFractionDigits: 0 })} ${currency}`;
   161→}
   162→
   163→function ChangeBadge({ value }: { value: number | null }) {
   164→  if (value === null) return <span className="text-muted-foreground text-xs">N/A</span>;
   165→  const isUp = value >= 0;
   166→  return (
   167→    <span
   168→      className={`inline-flex items-center gap-0.5 text-sm font-semibold ${isUp ? "text-emerald-500" : "text-red-500"}`}
   169→    >
   170→      {isUp ? <ArrowUpRight className="w-3.5 h-3.5" /> : <ArrowDownRight className="w-3.5 h-3.5" />}
   171→      {Math.abs(value).toFixed(2)}%
   172→    </span>
   173→  );
   174→}

   656→
   657→function LevelRow({
   658→  label,
   659→  value,
   660→  distance,
   661→  isDark,
   662→}: {
   663→  label: string;
   664→  value: number;
   665→  distance: number | null;
   666→  isDark: boolean;
   667→}) {
   668→  if (value <= 0) return null;
   669→  return (
   670→    <div className="flex items-center justify-between">
   671→      <span className={`text-xs ${isDark ? "text-zinc-400" : "text-zinc-500"}`}>
   672→        {label}
   673→      </span>
   674→      <div className="flex items-center gap-2">
   675→        <span className="text-xs font-mono">
   676→          {value.toLocaleString(undefined, { maximumFractionDigits: 0 })}
   677→        </span>
   678→        {distance !== null && (
   679→          <span
   680→            className={`text-xs font-semibold ${distance >= 0 ? "text-emerald-400" : "text-red-400"}`}
   681→          >
   682→            {distance >= 0 ? "+" : ""}
   683→            {distance.toFixed(1)}%
   684→          </span>
   685→        )}
   686→      </div>
   687→    </div>
   688→  );
   689→}
