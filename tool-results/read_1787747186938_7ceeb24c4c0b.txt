    24→def fmt_num(n, decimals=2):
    25→    """Format number with commas."""
    26→    if n is None:
    27→        return '—'
    28→    if abs(n) >= 1_000_000:
    29→        return f'{n:,.{decimals}f}'
    30→    if abs(n) >= 100:
    31→        return f'{n:,.{decimals}f}'
    32→    return f'{n:.{decimals}f}'
    33→
    34→
    35→def fmt_pct(n):
    36→    """Format percentage."""
    37→    if n is None:
    38→        return '—'
    39→    return f'{n * 100:.1f}%'
    40→
    41→
    42→def fmt_btc(n, decimals=8):
    43→    """Format BTC with smart decimal places (U6)."""
    44→    if n is None or n == 0:
    45→        return '0 BTC'
    46→    if abs(n) >= 1:
    47→        return f'{n:.4f} BTC'
    48→    if abs(n) >= 0.001:
    49→        return f'{n:.6f} BTC'
    50→    return f'{n:.8f} BTC'

    53→def color_for_value(val, thresholds):
    54→    """Return color class based on thresholds.
    55→    thresholds: [(threshold, color_class), ...] ascending order.
    56→    """
    57→    if val is None:
    58→        return 'neutral'
    59→    for threshold, color in thresholds:
    60→        if val < threshold:
    61→            return color
    62→    return thresholds[-1][1]
