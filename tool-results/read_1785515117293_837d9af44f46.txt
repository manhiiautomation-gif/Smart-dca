   475→    for accurate ROI of reserve-using strategies.
   476→    """
   477→    btc = 0.0
   478→    cash_reserve = 0.0
   479→    total_invested = 0.0
   480→    net_capital = 0.0       # Only new user capital (excludes reserve recycling)
   481→    total_sell_proceeds = 0.0
   482→    total_reserve_injected = 0.0
   483→    cooldown = 0
   484→    peak_value = 0.0
   485→    max_drawdown = 0.0
   486→    daily_log = []
   487→
   488→    for idx, row in df.iterrows():
   489→        price_thb = row['price_thb']
   490→        if cooldown > 0:
   491→            cooldown -= 1
   492→
   493→        state = {
   494→            'btc': btc, 'cash_reserve': cash_reserve,
   495→            'total_invested': total_invested, 'cooldown': cooldown,
   496→            'row': row, 'idx': idx
   497→        }
   498→        action = strategy_func(state)
   499→
   500→        buy_thb = action.get('buy_thb', 0)
   501→        sell_btc_pct = action.get('sell_btc_pct', 0)
   502→        sell_thb = action.get('sell_thb', 0)
   503→        to_reserve = action.get('to_reserve', 0)
   504→        sell_score = action.get('sell_score', 0)
   505→        reserve_injection = action.get('reserve_injection', 0)
   506→
   507→        actual_buy = apply_buy_fee(buy_thb)
   508→        btc_bought = actual_buy / price_thb if price_thb > 0 else 0
   509→        btc += btc_bought
   510→        total_invested += buy_thb
   511→        # Net capital: total buy minus what came from reserve
   512→        net_capital += buy_thb - reserve_injection
   513→        total_reserve_injected += reserve_injection
   514→
   515→        if sell_btc_pct > 0 and btc > 0:
   516→            btc_to_sell = btc * (sell_btc_pct / 100.0)
   517→            sell_proceeds = apply_sell_fee(btc_to_sell * price_thb)
   518→            btc -= btc_to_sell
   519→            cash_reserve += sell_proceeds
   520→            cooldown = action.get('new_cooldown', cooldown)
   521→
   522→        # THB-based selling (Style Beta v3 / Omega)
   523→        if sell_thb > 0 and btc > 0 and price_thb > 0:
   524→            btc_to_sell = sell_thb / price_thb
   525→            if btc_to_sell > btc:
   526→                btc_to_sell = btc
   527→            sell_proceeds = apply_sell_fee(btc_to_sell * price_thb)
   528→            btc -= btc_to_sell
   529→            cash_reserve += sell_proceeds
   530→            total_sell_proceeds += sell_proceeds
   531→            cooldown = action.get('new_cooldown', cooldown)
   532→
   533→        cash_reserve += to_reserve
   534→        cash_reserve = max(cash_reserve, 0.0)
   535→
   536→        portfolio_value = btc * price_thb + cash_reserve
   537→        avg_cost = total_invested / btc if btc > 0 else 0
   538→
   539→        if portfolio_value > peak_value:
   540→            peak_value = portfolio_value
   541→        if peak_value > 0:
   542→            drawdown = (peak_value - portfolio_value) / peak_value
   543→            if drawdown > max_drawdown:
   544→                max_drawdown = drawdown
   545→
   546→        daily_log.append({
   547→            'date': row['date'], 'price_thb': price_thb,
   548→            'btc': btc, 'cash_reserve': cash_reserve,
   549→            'total_invested': total_invested,
   550→            'portfolio_value': portfolio_value,
   551→            'avg_cost': avg_cost,
   552→            'max_drawdown_so_far': max_drawdown,
   553→        })
   554→
   555→    final_price = df.iloc[-1]['price_thb']
   556→    final_value = btc * final_price + cash_reserve
   557→    final_avg_cost = total_invested / btc if btc > 0 else 0
   558→    # ROI based on total invested (all THB through buy side)
   559→    roi_pct = ((final_value - total_invested) / total_invested * 100) if total_invested > 0 else 0
   560→    net_profit = final_value - total_invested
   561→    # True ROI: only new user capital (excludes reserve recycling)
   562→    true_roi_pct = ((final_value - net_capital) / net_capital * 100) if net_capital > 0 else 0
   563→    true_net_profit = final_value - net_capital
   564→
   565→    results = {
   566→        'strategy': strategy_name,
   567→        'total_invested': total_invested,
   568→        'net_capital': net_capital,
   569→        'total_btc': btc,
   570→        'avg_cost_thb': final_avg_cost,
   571→        'avg_cost_usd': final_avg_cost / USD_THB_RATE,
   572→        'final_value': final_value,
   573→        'cash_reserve': cash_reserve,
   574→        'roi_pct': roi_pct,
   575→        'true_roi_pct': true_roi_pct,
   576→        'net_profit': net_profit,
   577→        'true_net_profit': true_net_profit,
   578→        'total_sell_proceeds': total_sell_proceeds,
   579→        'total_reserve_injected': total_reserve_injected,
   580→        'max_drawdown_pct': max_drawdown * 100,
   581→    }
   582→    return results, pd.DataFrame(daily_log)
