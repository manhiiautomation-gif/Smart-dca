    49→    49→def fetch_binance_btc_price(days=2000):
    50→    50→    """
    51→    51→    Fetch REAL BTC daily close prices from Binance Spot API (klines).
    52→    52→    Uses pagination to retrieve up to 2000 days (~5.5 years) of ACTUAL
    53→    53→    historical exchange data. Contains ZERO random number generation.
    54→    54→    Returns DataFrame with ['date', 'price_usd'] or None on failure.
    55→    55→    """
    56→    56→    try:
    57→    57→        print("[DATA] Fetching REAL BTC prices from Binance API...")
    58→    58→        limit = 1000
    59→    59→        # Page 1: most recent 1000 days
    60→    60→        url1 = (f"https://api.binance.com/api/v3/klines?symbol=BTCUSDT"
    61→    61→               f"&interval=1d&limit={limit}")
    62→    62→        resp1 = requests.get(url1, timeout=15)
    63→    63→        if resp1.status_code != 200:
    64→    64→            print(f"[DATA] Binance returned status {resp1.status_code}.")
    65→    65→            return None
    66→    66→        candles1 = resp1.json()
    67→    67→        if not candles1:
    68→    68→            return None
    69→    69→        # Page 2: older 1000 days (paginate backwards)
    70→    70→        first_time = candles1[0][0]
    71→    71→        url2 = (f"https://api.binance.com/api/v3/klines?symbol=BTCUSDT"
    72→    72→               f"&interval=1d&limit={limit}&endTime={first_time - 1}")
    73→    73→        resp2 = requests.get(url2, timeout=15)
    74→    74→        candles2 = resp2.json() if resp2.status_code == 200 else []
    75→    75→
    76→    76→        all_candles = candles2 + candles1
    77→    77→        records = []
    78→    78→        for c in all_candles:
    79→    79→            dt = datetime.fromtimestamp(c[0] / 1000, tz=None).date()
    80→    80→            close_price = float(c[4])  # Index 4 = close price
    81→    81→            records.append({'date': dt, 'price_usd': close_price})
    82→    82→
    83→    83→        df = pd.DataFrame(records).drop_duplicates(subset='date').sort_values('date').reset_index(drop=True)
    84→    84→        print(f"[DATA] Binance returned {len(df)} days of REAL price data.")
    85→    85→        print(f"        Period: {df['date'].iloc[0]} to {df['date'].iloc[-1]}")
    86→    86→        return df
    87→    87→    except Exception as e:

    92→    92→def fetch_bgeometrics_metric(metric_name, token='7NqNRwWhyc'):
    93→    93→    """
    94→    94→    Fetch REAL on-chain metric from BGeometrics API.
    95→    95→    Includes retry on 429 rate-limit (up to 3 attempts with 5s delay).
    96→    96→    API returns JSON list with keys: 'd' (date string), metric_name (float).
    97→    97→    Returns DataFrame with ['date', metric_name] or None on failure.
    98→    98→    """
    99→    99→    url = f"https://api.bgeometrics.com/v1/{metric_name}?token={token}"
   100→   100→    for attempt in range(3):
   101→   101→        try:
   102→   102→            import time
   103→   103→            if attempt > 0:
   104→   104→                time.sleep(5)
   105→   105→            resp = requests.get(url, headers={'User-Agent': 'Mozilla/5.0'}, timeout=15)
   106→   106→            if resp.status_code == 429:
   107→   107→                print(f"[DATA] BGeometrics {metric_name}: HTTP 429 (attempt {attempt+1}/3)")
   108→   108→                continue
   109→   109→            if resp.status_code != 200:
   110→   110→                print(f"[DATA] BGeometrics {metric_name}: HTTP {resp.status_code}")
   111→   111→                return None
   112→   112→            data = resp.json()
   113→   113→            if not isinstance(data, list) or len(data) == 0:
   114→   114→                return None
   115→   115→            records = []
   116→   116→            for item in data:
   117→   117→                date_str = item.get('d')
   118→   118→                metric_val = item.get(metric_name)
   119→   119→                if date_str and metric_val is not None:
   120→   120→                    dt = pd.to_datetime(date_str).date()
   121→   121→                    records.append({'date': dt, metric_name: float(metric_val)})
   122→   122→            if not records:
   123→   123→                return None
   124→   124→            df = pd.DataFrame(records).drop_duplicates(subset='date').sort_values('date').reset_index(drop=True)
   125→   125→            print(f"[DATA] BGeometrics {metric_name}: {len(df)} records "
   126→   126→                  f"({df['date'].iloc[0]} to {df['date'].iloc[-1]})")
   127→   127→            return df
   128→   128→        except Exception as e:
   129→   129→            print(f"[DATA] BGeometrics {metric_name} fetch failed: {e}")
   130→   130→            if attempt == 2:
   131→   131→                return None
   132→   132→    return None

   135→   135→def generate_mock_btc_prices(start_date, end_date, seed=42):
   136→   136→    """
   137→   137→    Generate realistic mock BTC daily prices that simulate a full BTC cycle:
   138→   138→    Bull run -> Peak -> Bear market -> Recovery -> New Bull.
   139→   139→    This ensures selling mechanisms and drawdowns are properly tested.
   140→   140→    """
   141→   141→    np.random.seed(seed)
   142→   142→    dates = pd.date_range(start=start_date, end=end_date, freq='D')
   143→   143→    n = len(dates)
   144→   144→
   145→   145→    # Define waypoints (day_index, log_price) for a realistic BTC cycle:
   146→   146→    #   Day 0:     $30K (start)
   147→   147→    #   Day 200:   $55K (rally)
   148→   148→    #   Day 350:   $69K (cycle peak)
   149→   149→    #   Day 420:   $45K (initial crash)
   150→   150→    #   Day 600:   $22K (bear market low)
   151→   151→    #   Day 800:   $16K (capitulation bottom)
   152→   152→    #   Day 1000:  $25K (early recovery)
   153→   153→    #   Day 1200:  $42K (halving rally)
   154→   154→    #   Day 1400:  $65K (mid-bull)
   155→   155→    #   Day 1550:  $95K (acceleration)
   156→   156→    #   Day 1700:  $110K (new cycle peak)
   157→   157→    #   Day 1826:  $98K (slight pullback from peak)
   158→   158→    waypoints = [
   159→   159→        (0,     np.log(30000)),
   160→   160→        (200,   np.log(55000)),
   161→   161→        (350,   np.log(69000)),
   162→   162→        (420,   np.log(45000)),
   163→   163→        (600,   np.log(22000)),
   164→   164→        (800,   np.log(16000)),
   165→   165→        (1000,  np.log(25000)),
   166→   166→        (1200,  np.log(42000)),
   167→   167→        (1400,  np.log(65000)),
   168→   168→        (1550,  np.log(95000)),
   169→   169→        (1700,  np.log(110000)),
   170→   170→        (n - 1, np.log(98000)),
   171→   171→    ]
   172→   172→
   173→   173→    # Interpolate log-prices linearly between waypoints
   174→   174→    wp_days = [w[0] for w in waypoints]
   175→   175→    wp_logs = [w[1] for w in waypoints]
   176→   176→    all_days = np.arange(n)
   177→   177→    log_prices = np.interp(all_days, wp_days, wp_logs)

   183→   183→    log_prices += noise
   184→   184→
   185→   185→    # Light smoothing to avoid unrealistic daily jumps
   186→   186→    log_prices = pd.Series(log_prices).ewm(span=3, adjust=False).mean().values
   187→   187→
   188→   188→    prices = np.exp(log_prices)
   189→   189→    prices = np.clip(prices, 5000, 200000)
   190→   190→
   191→   191→    df = pd.DataFrame({'date': dates.date, 'price_usd': prices})
   192→   192→    print(f"[DATA] Generated {n} days of mock BTC prices ({start_date} to {end_date}).")
   193→   193→    print(f"        Price range: ${prices.min():,.0f} - ${prices.max():,.0f}")
   194→   194→    return df

   197→   197→def generate_mock_onchain_metrics(price_df, seed=123):
   198→   198→    """
   199→   199→    Generate realistic mock on-chain metrics (MVRV, NUPL, SOPR)
   200→   200→    correlated with BTC price movements.
   201→   201→    """
   202→   202→    np.random.seed(seed)
   203→   203→    n = len(price_df)
   204→   204→    dates = price_df['date'].values
   205→   205→    prices = price_df['price_usd'].values
   206→   206→    sma_365 = pd.Series(prices).rolling(365, min_periods=1).mean().values
   207→   207→
   208→   208→    # MVRV: Market Value to Realized Value (0.4 - 7.0 range)
   209→   209→    # MVRV should correlate strongly with price relative to realized value
   210→   210→    # and reach extreme highs (>3) during bull peaks and extreme lows (<0.7) in bears
   211→   211→    price_ratio = prices / np.maximum(sma_365, 1)
   212→   212→    cycle_phase = np.linspace(0, 3 * np.pi, n)  # ~1.5 full cycles over 5 years
   213→   213→
   214→   214→    # Base MVRV from price/SMA ratio, amplified to reach realistic extremes
   215→   215→    mvrv_base = np.power(price_ratio, 1.8)  # Power amplification for more extreme MVRV
   216→   216→    mvrv_cycle = 1.5 * np.sin(cycle_phase)  # Cyclical overlay for peaks/troughs
   217→   217→    mvrv_noise = np.random.normal(0, 0.08, n)
   218→   218→    mvrv = mvrv_base + mvrv_cycle + mvrv_noise
   219→   219→    mvrv = pd.Series(np.clip(mvrv, 0.4, 7.0)).ewm(span=14).mean().values
   220→   220→
   221→   221→    # NUPL: Net Unrealized Profit/Loss (-0.5 to 0.8 range)
   222→   222→    nupl_base = (prices - sma_365) / np.maximum(prices, 1)
   223→   223→    nupl = nupl_base * 2.5 + np.random.normal(0, 0.08, n)
   224→   224→    nupl = pd.Series(np.clip(nupl, -0.5, 0.8)).ewm(span=7).mean().values
   225→   225→
   226→   226→    # SOPR: Spent Output Profit Ratio (0.5 - 2.5, oscillates around 1.0)
   227→   227→    sopr_noise = np.random.normal(0, 0.05, n)
   228→   228→    sopr_trend = 1.0 + 0.3 * (price_ratio - 1.0) + np.sin(cycle_phase * 0.7) * 0.1
   229→   229→    sopr = pd.Series(np.clip(sopr_trend + sopr_noise, 0.5, 2.5)).ewm(span=7).mean().values
   230→   230→
   231→   231→    df = pd.DataFrame({'date': dates, 'mvrv': mvrv, 'nupl': nupl, 'sopr': sopr})
   232→   232→    print(f"[DATA] Generated {n} days of mock on-chain metrics.")
   233→   233→    print(f"        MVRV: {mvrv.min():.2f}-{mvrv.max():.2f}, NUPL: {nupl.min():.2f}-{nupl.max():.2f}, SOPR: {sopr.min():.2f}-{sopr.max():.2f}")
   234→   234→    return df

   237→   237→def compute_technical_indicators(df):
   238→   238→    """
   239→   239→    Add EMA 20, RSI 14, and SMA 365 to the DataFrame.
   240→   240→    """
   241→   241→    prices = df['price_usd']
   242→   242→    df['ema_20'] = prices.ewm(span=20, adjust=False).mean()
   243→   243→
   244→   244→    # RSI 14: Momentum oscillator (0-100)
   245→   245→    delta = prices.diff()
   246→   246→    gain = delta.where(delta > 0, 0.0).ewm(span=14, adjust=False).mean()
   247→   247→    loss = (-delta.where(delta < 0, 0.0)).ewm(span=14, adjust=False).mean()
   248→   248→    rs = gain / loss.replace(0, np.nan)
   249→   249→    df['rsi_14'] = 100 - (100 / (1 + rs))
   250→   250→    df['rsi_14'] = df['rsi_14'].fillna(50)
   251→   251→
   252→   252→    df['sma_365'] = prices.rolling(365, min_periods=1).mean()
   253→   253→    return df

   256→   256→def build_master_dataframe(years=5):
   257→   257→    """
   258→   258→    Main data pipeline: fetch REAL data from Binance + BGeometrics,
   259→   259→    merge into one master DataFrame. Falls back to proxy calculations for
   260→   260→    dates before BGeometrics data starts (before 2022-07-31).
   261→   261→    """
   262→   262→    # --- Step 1: REAL BTC Price Data from Binance ---
   263→   263→    price_df = fetch_binance_btc_price(days=2000)
   264→   264→    if price_df is None or len(price_df) < 365:
   265→   265→        print("[DATA] Binance failed. Using mock prices as LAST RESORT.")
   266→   266→        end_date = datetime.now().date()
   267→   267→        start_date = end_date - timedelta(days=int(years * 365.25))
   268→   268→        price_df = generate_mock_btc_prices(start_date, end_date)
   269→   269→
   270→   270→    # --- Step 2: REAL On-Chain Metrics from BGeometrics ---
   271→   271→    print("[DATA] Fetching REAL on-chain metrics from BGeometrics API...")
   272→   272→    mvrv_df = fetch_bgeometrics_metric('mvrv')
   273→   273→    nupl_df = fetch_bgeometrics_metric('nupl')
   274→   274→    sopr_df = fetch_bgeometrics_metric('sopr')
   275→   275→
   276→   276→    # Build on-chain DataFrame (real data where available, ensure columns exist)
   277→   277→    onchain_df = price_df[['date']].copy()
   278→   278→    onchain_df['mvrv'] = np.nan
   279→   279→    onchain_df['nupl'] = np.nan
   280→   280→    onchain_df['sopr'] = np.nan
   281→   281→    if mvrv_df is not None:
   282→   282→        onchain_df = onchain_df.drop(columns=['mvrv']).merge(mvrv_df, on='date', how='left')
   283→   283→        onchain_df['mvrv'] = onchain_df.get('mvrv', np.nan)
   284→   284→    if nupl_df is not None:
   285→   285→        onchain_df = onchain_df.drop(columns=['nupl']).merge(nupl_df, on='date', how='left')
   286→   286→        onchain_df['nupl'] = onchain_df.get('nupl', np.nan)
   287→   287→    if sopr_df is not None:
   288→   288→        onchain_df = onchain_df.drop(columns=['sopr']).merge(sopr_df, on='date', how='left')
   289→   289→        onchain_df['sopr'] = onchain_df.get('sopr', np.nan)
   290→   290→
   291→   291→    # --- Step 3: Merge Price + On-Chain ---
   292→   292→    master = price_df.merge(onchain_df, on='date', how='left')
   293→   293→    master = master.sort_values('date').reset_index(drop=True)
   294→   294→
   295→   295→    # --- Step 4: Forward-fill missing on-chain values (up to 2 consecutive days) ---
   296→   296→    for col in ['mvrv', 'nupl', 'sopr']:
   297→   297→        master[col] = master[col].ffill(limit=2)
   298→   298→
   299→   299→    # --- Step 5: Proxy fallback for dates before BGeometrics data ---
   300→   300→    # MVRV proxy = Price / 365-day SMA
   301→   301→    master['sma_365'] = master['price_usd'].rolling(365, min_periods=1).mean()
   302→   302→    master['mvrv_proxy'] = master['price_usd'] / master['sma_365']
   303→   303→    master['mvrv'] = master['mvrv'].fillna(master['mvrv_proxy'])
   304→   304→    # NUPL proxy = (Price - SMA365) / Price
   305→   305→    nupl_proxy = (master['price_usd'] - master['sma_365']) / master['price_usd']
   306→   306→    master['nupl'] = master['nupl'].fillna(nupl_proxy)
   307→   307→    # SOPR proxy = Price / EMA30
   308→   308→    ema30 = master['price_usd'].ewm(span=30, adjust=False).mean()
   309→   309→    sopr_proxy = master['price_usd'] / ema30
   310→   310→    master['sopr'] = master['sopr'].fillna(sopr_proxy)
   311→   311→
   312→   312→    # Report data coverage
   313→   313→    real_mvrv = master['mvrv'].notna().sum()
   314→   314→    real_nupl = master['nupl'].notna().sum()
   315→   315→    real_sopr = master['sopr'].notna().sum()
   316→   316→    print(f"\n[DATA] Data sources used:")
   317→   317→    print(f"        Price:  Binance REAL data ({len(master)} days)")
   318→   318→    print(f"        MVRV:   BGeometrics REAL ({(mvrv_df is not None)}), Proxy fill ({real_mvrv} total)")
   319→   319→    print(f"        NUPL:   BGeometrics REAL ({(nupl_df is not None)}), Proxy fill ({real_nupl} total)")
   320→   320→    print(f"        SOPR:   BGeometrics REAL ({(sopr_df is not None)}), Proxy fill ({real_sopr} total)")
   321→   321→
   322→   322→    # --- Step 6: Technical indicators ---
   323→   323→    master = compute_technical_indicators(master)
   324→   324→    master['price_thb'] = master['price_usd'] * USD_THB_RATE
   325→   325→
   326→   326→    print(f"\n[DATA] Master DataFrame ready: {len(master)} rows, {master['date'].min()} to {master['date'].max()}")
   327→   327→    return master

   344→   344→def backtest_strategy(df, strategy_func, strategy_name):
   345→   345→    """
   346→   346→    Generic backtest runner. Calls strategy_func for each day.
   347→   347→    strategy_func receives state dict and row, returns action dict.
   348→   348→    """
   349→   349→    btc = 0.0
   350→   350→    cash_reserve = 0.0
   351→   351→    total_invested = 0.0
   352→   352→    cooldown = 0
   353→   353→    peak_value = 0.0
   354→   354→    max_drawdown = 0.0
   355→   355→
   356→   356→    daily_log = []
   357→   357→
   358→   358→    for idx, row in df.iterrows():
   359→   359→        price_thb = row['price_thb']
   360→   360→
   361→   361→        if cooldown > 0:
   362→   362→            cooldown -= 1
   363→   363→
   364→   364→        state = {
   365→   365→            'btc': btc, 'cash_reserve': cash_reserve,
   366→   366→            'total_invested': total_invested, 'cooldown': cooldown,
   367→   367→            'row': row, 'idx': idx
   368→   368→        }
   369→   369→        action = strategy_func(state)
   370→   370→
   371→   371→        buy_thb = action.get('buy_thb', 0)
   372→   372→        sell_btc_pct = action.get('sell_btc_pct', 0)
   373→   373→        to_reserve = action.get('to_reserve', 0)
   374→   374→
   375→   375→        # Execute BUY
   376→   376→        actual_buy = apply_buy_fee(buy_thb)
   377→   377→        btc_bought = actual_buy / price_thb if price_thb > 0 else 0
   378→   378→        btc += btc_bought
   379→   379→        total_invested += buy_thb
   380→   380→
   381→   381→        # Execute SELL
   382→   382→        if sell_btc_pct > 0 and btc > 0:
   383→   383→            btc_to_sell = btc * (sell_btc_pct / 100.0)
   384→   384→            sell_proceeds = apply_sell_fee(btc_to_sell * price_thb)
   385→   385→            btc -= btc_to_sell
   386→   386→            cash_reserve += sell_proceeds
   387→   387→            cooldown = action.get('new_cooldown', cooldown)
   388→   388→
   389→   389→        # Reserve management
   390→   390→        cash_reserve += to_reserve
   391→   391→        cash_reserve = max(cash_reserve, 0.0)
   392→   392→
   393→   393→        # Portfolio valuation
   394→   394→        portfolio_value = btc * price_thb + cash_reserve
   395→   395→        avg_cost = total_invested / btc if btc > 0 else 0
   396→   396→
   397→   397→        # Max drawdown
   398→   398→        if portfolio_value > peak_value:
   399→   399→            peak_value = portfolio_value
   400→   400→        if peak_value > 0:
   401→   401→            drawdown = (peak_value - portfolio_value) / peak_value
   402→   402→            if drawdown > max_drawdown:
   403→   403→                max_drawdown = drawdown
   404→   404→
   405→   405→        daily_log.append({
   406→   406→            'date': row['date'], 'price_thb': price_thb,
   407→   407→            'btc': btc, 'cash_reserve': cash_reserve,
   408→   408→            'total_invested': total_invested,
   409→   409→            'portfolio_value': portfolio_value,
   410→   410→            'avg_cost': avg_cost,
   411→   411→            'max_drawdown_so_far': max_drawdown,
   412→   412→        })
   413→   413→
   414→   414→    final_price = df.iloc[-1]['price_thb']
   415→   415→    final_value = btc * final_price + cash_reserve
   416→   416→    final_avg_cost = total_invested / btc if btc > 0 else 0
   417→   417→    roi_pct = ((final_value - total_invested) / total_invested * 100) if total_invested > 0 else 0
   418→   418→    net_profit = final_value - total_invested
   419→   419→
   420→   420→    results = {
   421→   421→        'strategy': strategy_name,
   422→   422→        'total_invested': total_invested,
   423→   423→        'total_btc': btc,
   424→   424→        'avg_cost_thb': final_avg_cost,
   425→   425→        'avg_cost_usd': final_avg_cost / USD_THB_RATE,
   426→   426→        'final_value': final_value,
   427→   427→        'cash_reserve': cash_reserve,
   428→   428→        'roi_pct': roi_pct,
   429→   429→        'net_profit': net_profit,
   430→   430→        'max_drawdown_pct': max_drawdown * 100,
   431→   431→    }
   432→   432→    return results, pd.DataFrame(daily_log)
