    49→def fetch_coingecko_btc_price(days=1826):
    50→    """
    51→    Attempt to fetch BTC daily close prices from CoinGecko's free API.
    52→    Returns DataFrame with ['date', 'price_usd'] or None on failure.
    53→    """
    54→    url = (
    55→        f"https://api.coingecko.com/api/v3/coins/bitcoin/market_chart"
    56→        f"?vs_currency=usd&days={days}&interval=daily"
    57→    )
    58→    try:
    59→        print("[DATA] Fetching BTC prices from CoinGecko...")
    60→        resp = requests.get(url, headers={'accept': 'application/json'}, timeout=30)
    61→        if resp.status_code == 429:
    62→            print("[DATA] CoinGecko rate-limited. Will use mock data.")
    63→            return None
    64→        resp.raise_for_status()
    65→        data = resp.json()
    66→        prices = data.get('prices', [])
    67→        if not prices:
    68→            return None
    69→        records = []
    70→        for ts, price in prices:
    71→            dt = datetime.utcfromtimestamp(ts / 1000).date()
    72→            records.append({'date': dt, 'price_usd': price})
    73→        df = pd.DataFrame(records).drop_duplicates(subset='date').sort_values('date').reset_index(drop=True)
    74→        print(f"[DATA] CoinGecko returned {len(df)} days of price data.")
    75→        return df
    76→    except Exception as e:
    77→        print(f"[DATA] CoinGecko fetch failed: {e}. Will use mock data.")
    78→        return None

   117→def generate_mock_btc_prices(start_date, end_date, seed=42):
   118→    """
   119→    Generate realistic mock BTC daily prices that simulate a full BTC cycle:
   120→    Bull run -> Peak -> Bear market -> Recovery -> New Bull.
   121→    This ensures selling mechanisms and drawdowns are properly tested.
   122→    """
   123→    np.random.seed(seed)
   124→    dates = pd.date_range(start=start_date, end=end_date, freq='D')
   125→    n = len(dates)
   126→
   127→    # Define waypoints (day_index, log_price) for a realistic BTC cycle:
   128→    #   Day 0:     $30K (start)
   129→    #   Day 200:   $55K (rally)
   130→    #   Day 350:   $69K (cycle peak)
   131→    #   Day 420:   $45K (initial crash)
   132→    #   Day 600:   $22K (bear market low)
   133→    #   Day 800:   $16K (capitulation bottom)
   134→    #   Day 1000:  $25K (early recovery)
   135→    #   Day 1200:  $42K (halving rally)
   136→    #   Day 1400:  $65K (mid-bull)
   137→    #   Day 1550:  $95K (acceleration)
   138→    #   Day 1700:  $110K (new cycle peak)
   139→    #   Day 1826:  $98K (slight pullback from peak)
   140→    waypoints = [
   141→        (0,     np.log(30000)),
   142→        (200,   np.log(55000)),
   143→        (350,   np.log(69000)),
   144→        (420,   np.log(45000)),
   145→        (600,   np.log(22000)),
   146→        (800,   np.log(16000)),
   147→        (1000,  np.log(25000)),
   148→        (1200,  np.log(42000)),
   149→        (1400,  np.log(65000)),
   150→        (1550,  np.log(95000)),
   151→        (1700,  np.log(110000)),
   152→        (n - 1, np.log(98000)),
   153→    ]
   154→
   155→    # Interpolate log-prices linearly between waypoints
   156→    wp_days = [w[0] for w in waypoints]
   157→    wp_logs = [w[1] for w in waypoints]
   158→    all_days = np.arange(n)
   159→    log_prices = np.interp(all_days, wp_days, wp_logs)
   160→
   161→    # Add realistic GBM noise (50% annual vol — reduced from 65% to keep path near waypoints)
   162→    sigma = 0.50
   163→    dt = 1 / 365.25
   164→    noise = np.random.standard_normal(n) * sigma * np.sqrt(dt)
   165→    log_prices += noise
   166→
   167→    # Light smoothing to avoid unrealistic daily jumps
   168→    log_prices = pd.Series(log_prices).ewm(span=3, adjust=False).mean().values
   169→
   170→    prices = np.exp(log_prices)
   171→    prices = np.clip(prices, 5000, 200000)
   172→
   173→    df = pd.DataFrame({'date': dates.date, 'price_usd': prices})
   174→    print(f"[DATA] Generated {n} days of mock BTC prices ({start_date} to {end_date}).")
   175→    print(f"        Price range: ${prices.min():,.0f} - ${prices.max():,.0f}")
   176→    return df

   179→def generate_mock_onchain_metrics(price_df, seed=123):
   180→    """
   181→    Generate realistic mock on-chain metrics (MVRV, NUPL, SOPR)
   182→    correlated with BTC price movements.
   183→    """
   184→    np.random.seed(seed)
   185→    n = len(price_df)
   186→    dates = price_df['date'].values
   187→    prices = price_df['price_usd'].values
   188→    sma_365 = pd.Series(prices).rolling(365, min_periods=1).mean().values
   189→
   190→    # MVRV: Market Value to Realized Value (0.4 - 7.0 range)
   191→    # MVRV should correlate strongly with price relative to realized value
   192→    # and reach extreme highs (>3) during bull peaks and extreme lows (<0.7) in bears
   193→    price_ratio = prices / np.maximum(sma_365, 1)
   194→    cycle_phase = np.linspace(0, 3 * np.pi, n)  # ~1.5 full cycles over 5 years
   195→
   196→    # Base MVRV from price/SMA ratio, amplified to reach realistic extremes
   197→    mvrv_base = np.power(price_ratio, 1.8)  # Power amplification for more extreme MVRV
   198→    mvrv_cycle = 1.5 * np.sin(cycle_phase)  # Cyclical overlay for peaks/troughs
   199→    mvrv_noise = np.random.normal(0, 0.08, n)
   200→    mvrv = mvrv_base + mvrv_cycle + mvrv_noise
   201→    mvrv = pd.Series(np.clip(mvrv, 0.4, 7.0)).ewm(span=14).mean().values
   202→
   203→    # NUPL: Net Unrealized Profit/Loss (-0.5 to 0.8 range)
   204→    nupl_base = (prices - sma_365) / np.maximum(prices, 1)
   205→    nupl = nupl_base * 2.5 + np.random.normal(0, 0.08, n)
   206→    nupl = pd.Series(np.clip(nupl, -0.5, 0.8)).ewm(span=7).mean().values
   207→
   208→    # SOPR: Spent Output Profit Ratio (0.5 - 2.5, oscillates around 1.0)
   209→    sopr_noise = np.random.normal(0, 0.05, n)
   210→    sopr_trend = 1.0 + 0.3 * (price_ratio - 1.0) + np.sin(cycle_phase * 0.7) * 0.1
   211→    sopr = pd.Series(np.clip(sopr_trend + sopr_noise, 0.5, 2.5)).ewm(span=7).mean().values
   212→
   213→    df = pd.DataFrame({'date': dates, 'mvrv': mvrv, 'nupl': nupl, 'sopr': sopr})
   214→    print(f"[DATA] Generated {n} days of mock on-chain metrics.")
   215→    print(f"        MVRV: {mvrv.min():.2f}-{mvrv.max():.2f}, NUPL: {nupl.min():.2f}-{nupl.max():.2f}, SOPR: {sopr.min():.2f}-{sopr.max():.2f}")
   216→    return df

   219→def compute_technical_indicators(df):
   220→    """
   221→    Add EMA 20, RSI 14, and SMA 365 to the DataFrame.
   222→    """
   223→    prices = df['price_usd']
   224→    df['ema_20'] = prices.ewm(span=20, adjust=False).mean()
   225→
   226→    # RSI 14: Momentum oscillator (0-100)
   227→    delta = prices.diff()
   228→    gain = delta.where(delta > 0, 0.0).ewm(span=14, adjust=False).mean()
   229→    loss = (-delta.where(delta < 0, 0.0)).ewm(span=14, adjust=False).mean()
   230→    rs = gain / loss.replace(0, np.nan)
   231→    df['rsi_14'] = 100 - (100 / (1 + rs))
   232→    df['rsi_14'] = df['rsi_14'].fillna(50)
   233→
   234→    df['sma_365'] = prices.rolling(365, min_periods=1).mean()
   235→    return df

   238→def build_master_dataframe(years=5):
   239→    """
   240→    Main data pipeline: fetch or mock data, merge into one master DataFrame.
   241→    """
   242→    end_date = datetime.now().date()
   243→    start_date = end_date - timedelta(days=int(years * 365.25))
   244→
   245→    # Step 1: BTC Price Data
   246→    price_df = fetch_coingecko_btc_price(days=int(years * 365.25))
   247→    if price_df is None or len(price_df) < 365:
   248→        print("[DATA] Falling back to mock BTC prices...")
   249→        price_df = generate_mock_btc_prices(start_date, end_date)
   250→
   251→    # Step 2: On-Chain Metrics
   252→    mvrv_df = fetch_bgeometrics_metric('mvrv')
   253→    nupl_df = fetch_bgeometrics_metric('nupl')
   254→    sopr_df = fetch_bgeometrics_metric('sopr')
   255→
   256→    if (mvrv_df is None or nupl_df is None or sopr_df is None):
   257→        print("[DATA] Falling back to mock on-chain metrics...")
   258→        onchain_df = generate_mock_onchain_metrics(price_df)
   259→    else:
   260→        onchain_df = price_df[['date']].copy()
   261→        for metric_df, col in [(mvrv_df, 'mvrv'), (nupl_df, 'nupl'), (sopr_df, 'sopr')]:
   262→            onchain_df = onchain_df.merge(metric_df, on='date', how='left')
   263→
   264→    # Step 3: Merge
   265→    master = price_df.merge(onchain_df, on='date', how='left')
   266→    master = master.sort_values('date').reset_index(drop=True)
   267→
   268→    # Step 4: Forward-fill missing on-chain values (up to 2 consecutive days)
   269→    for col in ['mvrv', 'nupl', 'sopr']:
   270→        master[col] = master[col].ffill(limit=2)
   271→
   272→    # Step 5: Fallback proxy for still-missing MVRV
   273→    master['sma_365'] = master['price_usd'].rolling(365, min_periods=1).mean()
   274→    master['mvrv_proxy'] = master['price_usd'] / master['sma_365']
   275→    master['mvrv'] = master['mvrv'].fillna(master['mvrv_proxy'])
   276→    master['nupl'] = master['nupl'].fillna(0.0)
   277→    master['sopr'] = master['sopr'].fillna(1.0)
   278→
   279→    # Step 6: Technical indicators
   280→    master = compute_technical_indicators(master)
   281→    master['price_thb'] = master['price_usd'] * USD_THB_RATE
   282→
   283→    print(f"\n[DATA] Master DataFrame ready: {len(master)} rows, {master['date'].min()} to {master['date'].max()}")
   284→    return master

   291→def apply_buy_fee(thb_amount):
   292→    """Deduct 0.15% execution friction from a buy order."""
   293→    return thb_amount * (1 - BUY_FEE_PCT)
   294→

   296→def apply_sell_fee(thb_amount):
   297→    """Deduct 0.15% execution friction from a sell order."""
   298→    return thb_amount * (1 - SELL_FEE_PCT)

   301→def backtest_strategy(df, strategy_func, strategy_name):
   302→    """
   303→    Generic backtest runner. Calls strategy_func for each day.
   304→    strategy_func receives state dict and row, returns action dict.
   305→    """
   306→    btc = 0.0
   307→    cash_reserve = 0.0
   308→    total_invested = 0.0
   309→    cooldown = 0
   310→    peak_value = 0.0
   311→    max_drawdown = 0.0
   312→
   313→    daily_log = []
   314→
   315→    for idx, row in df.iterrows():
   316→        price_thb = row['price_thb']
   317→
   318→        if cooldown > 0:
   319→            cooldown -= 1
   320→
   321→        state = {
   322→            'btc': btc, 'cash_reserve': cash_reserve,
   323→            'total_invested': total_invested, 'cooldown': cooldown,
   324→            'row': row, 'idx': idx
   325→        }
   326→        action = strategy_func(state)
   327→
   328→        buy_thb = action.get('buy_thb', 0)
   329→        sell_btc_pct = action.get('sell_btc_pct', 0)
   330→        to_reserve = action.get('to_reserve', 0)
   331→
   332→        # Execute BUY
   333→        actual_buy = apply_buy_fee(buy_thb)
   334→        btc_bought = actual_buy / price_thb if price_thb > 0 else 0
   335→        btc += btc_bought
   336→        total_invested += buy_thb
   337→
   338→        # Execute SELL
   339→        if sell_btc_pct > 0 and btc > 0:
   340→            btc_to_sell = btc * (sell_btc_pct / 100.0)
   341→            sell_proceeds = apply_sell_fee(btc_to_sell * price_thb)
   342→            btc -= btc_to_sell
   343→            cash_reserve += sell_proceeds
   344→            cooldown = action.get('new_cooldown', cooldown)
   345→
   346→        # Reserve management
   347→        cash_reserve += to_reserve
   348→        cash_reserve = max(cash_reserve, 0.0)
   349→
   350→        # Portfolio valuation
   351→        portfolio_value = btc * price_thb + cash_reserve
   352→        avg_cost = total_invested / btc if btc > 0 else 0
   353→
   354→        # Max drawdown
   355→        if portfolio_value > peak_value:
   356→            peak_value = portfolio_value
   357→        if peak_value > 0:
   358→            drawdown = (peak_value - portfolio_value) / peak_value
   359→            if drawdown > max_drawdown:
   360→                max_drawdown = drawdown
   361→
   362→        daily_log.append({
   363→            'date': row['date'], 'price_thb': price_thb,
   364→            'btc': btc, 'cash_reserve': cash_reserve,
   365→            'total_invested': total_invested,
   366→            'portfolio_value': portfolio_value,
   367→            'avg_cost': avg_cost,
   368→            'max_drawdown_so_far': max_drawdown,
   369→        })
   370→
   371→    final_price = df.iloc[-1]['price_thb']
   372→    final_value = btc * final_price + cash_reserve
   373→    final_avg_cost = total_invested / btc if btc > 0 else 0
   374→    roi_pct = ((final_value - total_invested) / total_invested * 100) if total_invested > 0 else 0
   375→    net_profit = final_value - total_invested
   376→
   377→    results = {
   378→        'strategy': strategy_name,
   379→        'total_invested': total_invested,
   380→        'total_btc': btc,
   381→        'avg_cost_thb': final_avg_cost,
   382→        'avg_cost_usd': final_avg_cost / USD_THB_RATE,
   383→        'final_value': final_value,
   384→        'cash_reserve': cash_reserve,
   385→        'roi_pct': roi_pct,
   386→        'net_profit': net_profit,
   387→        'max_drawdown_pct': max_drawdown * 100,
   388→    }
   389→    return results, pd.DataFrame(daily_log)

   472→# --- STRATEGY 4: STYLE G v2 (Adaptive Hybrid Flagship) ---
   473→def strategy_style_g_v2(df_precomputed):
   474→    """
   475→    Factory function. Precomputes 730-day rolling percentiles for MVRV/NUPL,
   476→    then returns the actual strategy function. Uses Composite Value Score (S_t).
   477→    """
   478→    mvrv_pct = df_precomputed['mvrv'].rolling(730, min_periods=30).apply(
   479→        lambda x: pd.Series(x).rank(pct=True).iloc[-1] * 100, raw=False
   480→    ).values
   481→    nupl_pct = df_precomputed['nupl'].rolling(730, min_periods=30).apply(
   482→        lambda x: pd.Series(x).rank(pct=True).iloc[-1] * 100, raw=False
   483→    ).values
   484→
   485→    def strategy_func(state):
   486→        row = state['row']
   487→        idx = state['idx']
   488→        cash = state['cash_reserve']
   489→        cooldown = state['cooldown']
   490→        mvrv = row['mvrv']
   491→        nupl = row['nupl']
   492→        sopr = row['sopr']
   493→        rsi = row['rsi_14']
   494→        ema20 = row['ema_20']
   495→        price = row['price_usd']
   496→
   497→        m_pct = mvrv_pct[idx] if not np.isnan(mvrv_pct[idx]) else 50
   498→        n_pct = nupl_pct[idx] if not np.isnan(nupl_pct[idx]) else 50
   499→
   500→        # SOPR Score
   501→        if sopr < 0.95:
   502→            sopr_score = 100
   503→        elif sopr < 0.98:
   504→            sopr_score = 75
   505→        elif sopr < 1.00:
   506→            sopr_score = 50
   507→        else:
   508→            sopr_score = 20
   509→
   510→        # Composite Value Score S_t (0-100, higher = more undervalued)
   511→        s_t = 0.45 * (100 - m_pct) + 0.35 * (100 - n_pct) + 0.20 * sopr_score
   512→
   513→        # Out-of-Pocket Multiplier
   514→        if s_t < 20:
   515→            oop_mult = 0.0
   516→        elif s_t < 40:
   517→            oop_mult = 0.5
   518→        elif s_t < 70:
   519→            oop_mult = 1.0
   520→        elif s_t < 85:
   521→            oop_mult = 2.0
   522→        elif s_t < 92:
   523→            oop_mult = 3.5
   524→        else:
   525→            oop_mult = 5.0
   526→
   527→        buy_amount = BASE_BUDGET_THB * oop_mult
   528→
   529→        # Unspent budget to reserve
   530→        to_reserve = BASE_BUDGET_THB * (1.0 - oop_mult) if oop_mult < 1.0 else 0.0
   531→
   532→        # Drip injection
   533→        injection = 0.0

   538→                injection = cash * 0.08

   540→                injection = cash * 0.06
   541→            injection = min(injection, 600)
   542→            buy_amount += injection
   543→            to_reserve -= injection
   544→
   545→        # Two-stage top skim
   546→        sell_pct = 0.0
   547→        new_cooldown = cooldown
   548→        if cooldown == 0 and state['btc'] > 0:
   549→            if m_pct >= 90:
   550→                sell_pct = 12.0
   551→                new_cooldown = 45
   552→            elif s_t < 15 and price < ema20 and rsi < 68:

   556→        return {
   557→            'buy_thb': buy_amount, 'sell_btc_pct': sell_pct,

   564→# --- STRATEGY 5: STYLE ALPHA (Innovated) ---
   565→def strategy_style_alpha(df_precomputed):
   566→    """
   567→    SMART DCA STYLE ALPHA — Adaptive Volatility-Regime Strategy
   568→
   569→    DESIGN PHILOSOPHY — Fixes structural weaknesses of Styles C, E, G v2:
   570→
   571→    FIX 1 (vs Style C): Adds multi-stage fractional scale-out selling.
   572→      Style C never sells, so it suffers full drawdown exposure.
   573→      Alpha sells in gradual tranches (5% -> 10% -> 15%) based on severity.
   574→
   575→    FIX 2 (vs Style E): Uses 200-day SMA regime filter.
   576→      Style E sells too early in extended bull runs (MVRV > 3.0 can fire mid-bull).
   577→      Alpha raises sell thresholds in confirmed bull regimes.
   578→      Also uses adaptive cooldown (60-90 days) scaling with volatility.
   579→
   580→    FIX 3 (vs Style G v2): Risk Parity reserve management (12% target).
   581→      G v2 suffers cash drag — too much idle capital in reserve.
   582→      Alpha targets a 12% reserve-to-portfolio ratio and injects surplus aggressively
   583→      (25% of surplus, cap 1200 THB/day). Uses 365-day lookback for faster adaptation.
   584→
   585→    INNOVATIONS:
   586→    1. Volatility-Adjusted Sizing: Buy amounts divided by normalized vol score
   587→       (baseline 1.0, clamped 0.5-1.5). Softer than initial design to preserve
   588→       bottom-buying aggressiveness.
   589→    2. Regime Filter (200-day SMA): Bull vs Bear changes sell thresholds and
   590→       reserve deployment rate.
   591→    3. Fractional Scale-Out: Small, conservative sell tranches (3%/5%/8%)
   592→       with long adaptive cooldowns (75-105 days) to preserve BTC exposure.
   593→    4. Capitulation Booster: Matches Style C's SOPR < 0.95 logic with even
   594→       more aggressive multiplier (5.0-5.5x) during extreme fear.
   595→    """
   596→    # Precompute 365-day rolling percentiles (faster than G v2's 730-day)
   597→    mvrv_pct = df_precomputed['mvrv'].rolling(365, min_periods=30).apply(
   598→        lambda x: pd.Series(x).rank(pct=True).iloc[-1] * 100, raw=False
   599→    ).values
   600→    nupl_pct = df_precomputed['nupl'].rolling(365, min_periods=30).apply(
   601→        lambda x: pd.Series(x).rank(pct=True).iloc[-1] * 100, raw=False
   602→    ).values
   603→
   604→    # 30-day annualized realized volatility
   605→    returns = df_precomputed['price_usd'].pct_change()
   606→    vol_30 = (returns.rolling(30, min_periods=5).std() * np.sqrt(365)).fillna(0.60).values
   607→
   608→    # 200-day SMA for regime detection
   609→    sma_200 = df_precomputed['price_usd'].rolling(200, min_periods=50).mean().values
   610→
   611→    # Precompute all-time high (ATH) tracking for additional sell signal
   612→    cummax_price = pd.Series(df_precomputed['price_usd']).cummax().values
   613→
   614→    def strategy_func(state):
   615→        row = state['row']
   616→        idx = state['idx']
   617→        cash = state['cash_reserve']
   618→        cooldown = state['cooldown']
   619→        btc = state['btc']
   620→        price_usd = row['price_usd']
   621→        price_thb = row['price_thb']
   622→        mvrv = row['mvrv']
   623→        sopr = row['sopr']
   624→        rsi = row['rsi_14']
   625→
   626→        # 1. REGIME: Bull if price > 200-day SMA
   627→        s200 = sma_200[idx] if idx < len(sma_200) else price_usd
   628→        is_bull = price_usd > s200 if not np.isnan(s200) else True
   629→        regime_score = 20 if is_bull else 80
   630→
   631→        # 2. VOLATILITY SCORING (normalized to 1.0 baseline, clamped 0.5-1.5)
   632→        # Softer vol adjustment: only mild sizing during high/low vol
   633→        vol = vol_30[idx] if idx < len(vol_30) else 0.60
   634→        vol_score = np.clip(vol / 1.0, 0.5, 1.5)
   635→
   636→        # 3. SOPR SCORE
   637→        if sopr < 0.95:
   638→            sopr_score = 100
   639→        elif sopr < 0.98:
   640→            sopr_score = 75
   641→        elif sopr < 1.00:
   642→            sopr_score = 50
   643→        else:
   644→            sopr_score = 20
   645→
   646→        # 4. VALUATION SCORE V_t (0-100, higher = more undervalued)
   647→        m_pct = mvrv_pct[idx] if not np.isnan(mvrv_pct[idx]) else 50
   648→        n_pct = nupl_pct[idx] if not np.isnan(nupl_pct[idx]) else 50
   649→        v_t = 0.40 * (100 - m_pct) + 0.30 * (100 - n_pct) + 0.15 * sopr_score + 0.15 * regime_score
   650→
   651→        # 5. BUY MULTIPLIER (volatility-adjusted, with capitulation booster)
   652→        if v_t < 15:
   653→            base_mult = 0.0
   654→        elif v_t < 30:
   655→            base_mult = 0.5
   656→        elif v_t < 55:
   657→            base_mult = 1.0
   658→        elif v_t < 75:
   659→            base_mult = 2.0
   660→        elif v_t < 90:
   661→            base_mult = 3.5
   662→        else:
   663→            base_mult = 5.0
   664→
   665→        # Capitulation booster: if SOPR < 0.95 AND MVRV < 0.8, boost aggressively
   666→        # (This matches Style C's booster logic but with even more aggression)
   667→        if mvrv < 0.8 and sopr < 0.95:
   668→            base_mult = max(base_mult, 5.5)
   669→        elif mvrv < 1.0 and sopr < 0.95:
   670→            base_mult = max(base_mult, 5.0)
   671→
   672→        # Divide by vol_score: buy LESS when vol is extreme (but softer adjustment)
   673→        effective_mult = base_mult / vol_score
   674→        buy_amount = BASE_BUDGET_THB * effective_mult
   675→
   676→        # 6. RESERVE MANAGEMENT (Risk Parity, target 12% of portfolio)
   677→        # Lower target than before to reduce cash drag
   678→        to_reserve = 0.0
   679→        portfolio_val = btc * price_thb + cash
   680→        if portfolio_val > 0:
   681→            target_reserve = portfolio_val * 0.12
   682→            # Inject surplus when reserve is too large (aggressive deployment)
   683→            if cash > target_reserve * 1.8 and v_t > 35:
   684→                surplus = cash - target_reserve * 1.8
   685→                injection = min(surplus * 0.25, 1200)  # 25% of surplus, cap 1200 THB
   686→                buy_amount += injection
   687→                to_reserve -= injection
   688→            # Fund reserve when too small and market is expensive
   689→            elif cash < target_reserve * 0.3 and v_t < 25:
   690→                reserve_fund = BASE_BUDGET_THB * 0.4
   691→                to_reserve = reserve_fund
   692→                buy_amount = max(0, buy_amount - reserve_fund)
   693→
   694→        buy_amount = max(0, buy_amount)
   695→
   696→        # 7. FRACTIONAL SCALE-OUT (progressive selling, cooldown == 0)
   697→        # Only sells a SMALL portion to preserve BTC exposure
   698→        sell_pct = 0.0
   699→        new_cooldown = cooldown
   700→        if cooldown == 0 and btc > 0:
   701→            ath = cummax_price[idx] if idx < len(cummax_price) else price_usd
   702→            drawdown_from_ath = (ath - price_usd) / ath if ath > 0 else 0
   703→
   704→            if is_bull:
   705→                # BULL: Very high thresholds — only trim at extreme euphoria
   706→                if v_t < 8 and rsi > 78:
   707→                    sell_pct = 3.0  # Just 3% — preserve upside
   708→                elif v_t < 5 and rsi > 82:
   709→                    sell_pct = 5.0  # 5% at extreme euphoria
   710→            else:
   711→                # BEAR: Defensive — take some profits when MVRV is historically extreme
   712→                if m_pct > 93:
   713→                    sell_pct = 3.0
   714→                elif m_pct > 97:
   715→                    sell_pct = 5.0
   716→                elif mvrv > 4.0:
   717→                    sell_pct = 8.0  # Only 8% at extreme overvaluation
   718→            if sell_pct > 0:
   719→                # Adaptive cooldown: 75-105 days (longer to avoid whipsaw)
   720→                new_cooldown = 75 + int(vol_score * 20)
   721→
   722→        return {
   723→            'buy_thb': buy_amount, 'sell_btc_pct': sell_pct,
   724→            'to_reserve': to_reserve, 'new_cooldown': new_cooldown,
   725→        }
   726→
   727→    return strategy_func
