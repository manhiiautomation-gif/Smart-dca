    61→    61→def _cache_path(name):
    62→    62→    """Return CSV cache file path for a given data source."""
    63→    63→    return os.path.join(CACHE_DIR, f'{name}.csv')

    69→    69→def _load_cache(name):
    70→    70→    """Load DataFrame from CSV cache if it exists and is recent (< 7 days)."""
    71→    71→    path = _cache_path(name)
    72→    72→    if os.path.exists(path):
    73→    73→        age_hours = (time.time() - os.path.getmtime(path)) / 3600
    74→    74→        if age_hours < CACHE_MAX_AGE_HOURS:
    75→    75→            df = pd.read_csv(path, parse_dates=['date'])
    76→    76→            df['date'] = df['date'].dt.date
    77→    77→            print(f"[CACHE] Loaded {name} from disk ({len(df)} rows, {age_hours:.1f}h old)")
    78→    78→            return df
    79→    79→        else:
    80→    80→            print(f"[CACHE] {name} cache expired ({age_hours:.1f}h old), re-fetching...")
    81→    81→    return None

   142→   142→def fetch_coinmetrics_mvrv(start_date, end_date):
   143→   143→    """
   144→   144→    Fetch REAL MVRV from CoinMetrics Community API (FREE, no API key needed).
   145→   145→    Metric: CapMVRVCur (Market Value to Realized Value Ratio).
   146→   146→    High rate limit — no issues in practice.
   147→   147→    Cached to CSV — subsequent runs skip the API entirely.
   148→   148→    Returns DataFrame with ['date', 'mvrv'] or None on failure.
   149→   149→    """
   150→   150→    cache_key = 'coinmetrics_mvrv'
   151→   151→    cached = _load_cache(cache_key)
   152→   152→    if cached is not None:
   153→   153→        return cached
   154→   154→
   155→   155→    try:
   156→   156→        print(f"[DATA] Fetching REAL MVRV from CoinMetrics Community API...")
   157→   157→        all_records = []
   158→   158→        # CoinMetrics limits ~100 rows per request — use 90-day chunks
   159→   159→        current_start = start_date.strftime('%Y-%m-%dT00:00:00Z')
   160→   160→        end_str = end_date.strftime('%Y-%m-%dT23:59:59Z')
   161→   161→        
   162→   162→        while current_start < end_str:
   163→   163→            chunk_end_dt = (pd.to_datetime(current_start) + timedelta(days=90)).strftime('%Y-%m-%dT00:00:00Z')
   164→   164→            if chunk_end_dt > end_str:
   165→   165→                chunk_end_dt = end_str
   166→   166→            
   167→   167→            url = "https://community-api.coinmetrics.io/v4/timeseries/asset-metrics"
   168→   168→            params = {
   169→   169→                'assets': 'btc',
   170→   170→                'metrics': 'CapMVRVCur',
   171→   171→                'frequency': '1d',
   172→   172→                'start_time': current_start,
   173→   173→                'end_time': chunk_end_dt,
   174→   174→            }
   175→   175→            resp = requests.get(url, params=params, timeout=30)
   176→   176→            if resp.status_code != 200:
   177→   177→                print(f"[DATA] CoinMetrics HTTP {resp.status_code}: {resp.text[:200]}")
   178→   178→                if not all_records:
   179→   179→                    return None
   180→   180→                break
   181→   181→            
   182→   182→            data = resp.json().get('data', [])
   183→   183→            for item in data:
   184→   184→                dt = pd.to_datetime(item['time']).date()
   185→   185→                val = float(item['CapMVRVCur'])
   186→   186→                if val > 0:  # skip invalid MVRV
   187→   187→                    all_records.append({'date': dt, 'mvrv': val})
   188→   188→            
   189→   189→            print(f"[DATA]   {current_start[:10]} to {chunk_end_dt[:10]}: {len(data)} rows")
   190→   190→            current_start = chunk_end_dt
   191→   191→            time.sleep(0.2)  # CoinMetrics has high rate limits
   192→   192→        
   193→   193→        if not all_records:
   194→   194→            return None
   195→   195→        
   196→   196→        df = pd.DataFrame(all_records).drop_duplicates(subset='date').sort_values('date').reset_index(drop=True)
   197→   197→        print(f"[DATA] CoinMetrics MVRV: {len(df)} records ({df['date'].iloc[0]} to {df['date'].iloc[-1]})")
   198→   198→        print(f"        MVRV range: {df['mvrv'].min():.3f} - {df['mvrv'].max():.3f}")
   199→   199→        _save_cache(cache_key, df)

   202→   202→        print(f"[DATA] CoinMetrics MVRV fetch failed: {e}")

   206→   206→def fetch_bgeometrics_metric(metric_name, token='7NqNRwWhyc'):
   207→   207→    """
   208→   208→    Fetch REAL on-chain metric from BGeometrics API (FALLBACK only).
   209→   209→    Used for SOPR which CoinMetrics doesn't offer for free.
   210→   210→    """
   211→   211→    cache_key = f'bgeometrics_{metric_name}'
   212→   212→    cached = _load_cache(cache_key)
   213→   213→    if cached is not None:
   214→   214→        return cached
   215→   215→
   216→   216→    url = f"https://api.bgeometrics.com/v1/{metric_name}?token={token}"
   217→   217→    for attempt in range(3):
   218→   218→        try:
   219→   219→            if attempt > 0:
   220→   220→                wait = 10 * attempt
   221→   221→                print(f"[DATA] BGeometrics {metric_name}: retry in {wait}s (attempt {attempt+1}/3)")
   222→   222→                time.sleep(wait)
   223→   223→            resp = requests.get(url, headers={'User-Agent': 'Mozilla/5.0'}, timeout=15)
   224→   224→            if resp.status_code == 429:
   225→   225→                print(f"[DATA] BGeometrics {metric_name}: HTTP 429 rate-limit (attempt {attempt+1}/3)")
   226→   226→                continue
   227→   227→            if resp.status_code != 200:
   228→   228→                return None
   229→   229→            data = resp.json()
   230→   230→            if not isinstance(data, list) or len(data) == 0:
   231→   231→                return None
   232→   232→            records = []
   233→   233→            for item in data:
   234→   234→                date_str = item.get('d')
   235→   235→                metric_val = item.get(metric_name)
   236→   236→                if date_str and metric_val is not None:
   237→   237→                    dt = pd.to_datetime(date_str).date()
   238→   238→                    records.append({'date': dt, metric_name: float(metric_val)})
   239→   239→            if not records:
   240→   240→                return None
   241→   241→            df = pd.DataFrame(records).drop_duplicates(subset='date').sort_values('date').reset_index(drop=True)
   242→   242→            print(f"[DATA] BGeometrics {metric_name}: {len(df)} records "
   243→   243→                  f"({df['date'].iloc[0]} to {df['date'].iloc[-1]})")
   244→   244→            _save_cache(cache_key, df)
   245→   245→            return df
   246→   246→        except Exception as e:
   247→   247→            if attempt == 2:
   248→   248→                return None
   249→   249→    return None

   289→   289→    sma_365 = pd.Series(prices).rolling(365, min_periods=1).mean().values
   290→   290→    price_ratio = prices / np.maximum(sma_365, 1)
   291→   291→    cycle_phase = np.linspace(0, 3 * np.pi, n)
   292→   292→    mvrv_base = np.power(price_ratio, 1.8)

   295→   295→    mvrv = mvrv_base + mvrv_cycle + mvrv_noise
   296→   296→    mvrv = pd.Series(np.clip(mvrv, 0.4, 7.0)).ewm(span=14).mean().values
   297→   297→    nupl_base = (prices - sma_365) / np.maximum(prices, 1)
   298→   298→    nupl = nupl_base * 2.5 + np.random.normal(0, 0.08, n)
   299→   299→    nupl = pd.Series(np.clip(nupl, -0.5, 0.8)).ewm(span=7).mean().values

   308→   308→def compute_technical_indicators(df):
   309→   309→    """Add EMA 20, RSI 14, and SMA 365 to the DataFrame."""
   310→   310→    prices = df['price_usd']
   311→   311→    df['ema_20'] = prices.ewm(span=20, adjust=False).mean()
   312→   312→    delta = prices.diff()
   313→   313→    gain = delta.where(delta > 0, 0.0).ewm(span=14, adjust=False).mean()
   314→   314→    loss = (-delta.where(delta < 0, 0.0)).ewm(span=14, adjust=False).mean()
   315→   315→    rs = gain / loss.replace(0, np.nan)
   316→   316→    df['rsi_14'] = 100 - (100 / (1 + rs))
   317→   317→    df['rsi_14'] = df['rsi_14'].fillna(50)
   318→   318→    df['sma_365'] = prices.rolling(365, min_periods=1).mean()
   319→   319→    return df

   322→   322→def build_master_dataframe(years=5):
   323→   323→    """
   324→   324→    Main data pipeline: fetch REAL data (with CSV cache), merge into master DF.
   325→   325→    Falls back to mock data ONLY if both API and cache fail.
   326→   326→    Falls back to proxy calculations for dates before BGeometrics data starts.
   327→   327→    """
   328→   328→    used_cache = {'price': False, 'mvrv': False, 'nupl': False, 'sopr': False}
   329→   329→
   330→   330→    # --- Step 1: REAL BTC Price from Binance (or cache) ---
   331→   331→    price_df = fetch_binance_btc_price(days=2000)
   332→   332→    if price_df is None or len(price_df) < 365:
   333→   333→        # Check if there's an old cache we can still use
   334→   334→        old_cache = pd.read_csv(_cache_path('binance_btc_prices'), parse_dates=['date'])
   335→   335→        if len(old_cache) >= 365:
   336→   336→            old_cache['date'] = old_cache['date'].dt.date
   337→   337→            price_df = old_cache
   338→   338→            print("[DATA] Using expired cache as fallback for prices.")
   339→   339→        else:
   340→   340→            print("[DATA] Binance FAILED. Using mock prices as LAST RESORT.")
   341→   341→            end_date = datetime.now().date()
   342→   342→            start_date = end_date - timedelta(days=int(years * 365.25))
   343→   343→            price_df = generate_mock_btc_prices(start_date, end_date)
   344→   344→    else:
   345→   345→        used_cache['price'] = os.path.exists(_cache_path('binance_btc_prices'))
   346→   346→
   347→   347→    # --- Step 2: REAL On-Chain Metrics ---
   348→   348→    # PRIMARY:   CoinMetrics Community API (CapMVRVCur) — FREE, high rate limit
   349→   349→    # DERIVED:   NUPL = 1 - (1/MVRV) — mathematically identical to on-chain NUPL
   350→   350→    # FALLBACK:  BGeometrics SOPR (rate-limited) → Proxy (Price/EMA30)
   351→   351→    data_start = price_df['date'].iloc[0]
   352→   352→    data_end = price_df['date'].iloc[-1]
   353→   353→
   354→   354→    print(f"\n[DATA] === ON-CHAIN DATA PIPELINE ===")
   355→   355→    print(f"        Primary source: CoinMetrics Community API (CapMVRVCur)")
   356→   356→    print(f"        NUPL: Derived from MVRV (NUPL = 1 - 1/MVRV)")
   357→   357→    print(f"        SOPR: BGeometrics fallback -> Proxy (Price/EMA30)")
   358→   358→
   359→   359→    # Fetch REAL MVRV from CoinMetrics
   360→   360→    mvrv_df = fetch_coinmetrics_mvrv(data_start, data_end)
   361→   361→    nupl_df = None  # Will be derived from MVRV
   362→   362→
   363→   363→    # Try BGeometrics SOPR (non-critical, proxy is acceptable)
   364→   364→    time.sleep(2)
   365→   365→    sopr_df = fetch_bgeometrics_metric('sopr')
   366→   366→
   367→   367→    # Merge on-chain data
   368→   368→    onchain_df = price_df[['date']].copy()
   369→   369→    onchain_df['mvrv'] = np.nan
   370→   370→    onchain_df['nupl'] = np.nan
   371→   371→    onchain_df['sopr'] = np.nan
   372→   372→    if mvrv_df is not None:
   373→   373→        onchain_df = onchain_df.drop(columns=['mvrv']).merge(mvrv_df, on='date', how='left')
   374→   374→    if sopr_df is not None:
   375→   375→        onchain_df = onchain_df.drop(columns=['sopr']).merge(sopr_df, on='date', how='left')
   376→   376→
   377→   377→    master = price_df.merge(onchain_df, on='date', how='left')
   378→   378→    master = master.sort_values('date').reset_index(drop=True)
   379→   379→
   380→   380→    # Forward-fill missing on-chain (up to 2 days)
   381→   381→    for col in ['mvrv', 'nupl', 'sopr']:
   382→   382→        master[col] = master[col].ffill(limit=2)
   383→   383→
   384→   384→    # --- Step 3: Proxy fallback ---
   385→   385→    master['sma_365'] = master['price_usd'].rolling(365, min_periods=1).mean()
   386→   386→    # MVRV proxy (only if CoinMetrics failed)
   387→   387→    master['mvrv_proxy'] = master['price_usd'] / master['sma_365']
   388→   388→    master['mvrv'] = master['mvrv'].fillna(master['mvrv_proxy'])
   389→   389→
   390→   390→    # NUPL: DERIVE from real MVRV where available, proxy otherwise
   391→   391→    # Real: NUPL = 1 - (1/MVRV)   [on-chain identity]
   392→   392→    # Proxy: NUPL = (Price - SMA365) / Price

   446→   446→# SECTION 2: BACKTEST ENGINE
   447→   447→# ============================================================
   448→   448→
   449→   449→def apply_buy_fee(thb_amount):
   450→   450→    return thb_amount * (1 - BUY_FEE_PCT)
   451→   451→
   452→   452→def apply_sell_fee(thb_amount):
   453→   453→    return thb_amount * (1 - SELL_FEE_PCT)
   454→   454→
   455→   455→def backtest_strategy(df, strategy_func, strategy_name):
   456→   456→    """
   457→   457→    Generic backtest runner. Calls strategy_func for each day.
   458→   458→    """
   459→   459→    btc = 0.0
   460→   460→    cash_reserve = 0.0
   461→   461→    total_invested = 0.0
   462→   462→    cooldown = 0
   463→   463→    peak_value = 0.0
   464→   464→    max_drawdown = 0.0
   465→   465→    daily_log = []
   466→   466→
   467→   467→    for idx, row in df.iterrows():
   468→   468→        price_thb = row['price_thb']
   469→   469→        if cooldown > 0:
   470→   470→            cooldown -= 1
   471→   471→
   472→   472→        state = {
   473→   473→            'btc': btc, 'cash_reserve': cash_reserve,
   474→   474→            'total_invested': total_invested, 'cooldown': cooldown,
   475→   475→            'row': row, 'idx': idx
   476→   476→        }
   477→   477→        action = strategy_func(state)
   478→   478→
   479→   479→        buy_thb = action.get('buy_thb', 0)
   480→   480→        sell_btc_pct = action.get('sell_btc_pct', 0)
   481→   481→        to_reserve = action.get('to_reserve', 0)
   482→   482→
   483→   483→        actual_buy = apply_buy_fee(buy_thb)
   484→   484→        btc_bought = actual_buy / price_thb if price_thb > 0 else 0
   485→   485→        btc += btc_bought
   486→   486→        total_invested += buy_thb
   487→   487→
   488→   488→        if sell_btc_pct > 0 and btc > 0:
   489→   489→            btc_to_sell = btc * (sell_btc_pct / 100.0)
   490→   490→            sell_proceeds = apply_sell_fee(btc_to_sell * price_thb)
   491→   491→            btc -= btc_to_sell
   492→   492→            cash_reserve += sell_proceeds
   493→   493→            cooldown = action.get('new_cooldown', cooldown)
   494→   494→
   495→   495→        cash_reserve += to_reserve
   496→   496→        cash_reserve = max(cash_reserve, 0.0)
   497→   497→
   498→   498→        portfolio_value = btc * price_thb + cash_reserve
   499→   499→        avg_cost = total_invested / btc if btc > 0 else 0
   500→   500→
   501→   501→        if portfolio_value > peak_value:
   502→   502→            peak_value = portfolio_value
   503→   503→        if peak_value > 0:
   504→   504→            drawdown = (peak_value - portfolio_value) / peak_value
   505→   505→            if drawdown > max_drawdown:
   506→   506→                max_drawdown = drawdown
   507→   507→
   508→   508→        daily_log.append({
   509→   509→            'date': row['date'], 'price_thb': price_thb,
   510→   510→            'btc': btc, 'cash_reserve': cash_reserve,
   511→   511→            'total_invested': total_invested,
   512→   512→            'portfolio_value': portfolio_value,
   513→   513→            'avg_cost': avg_cost,
   514→   514→            'max_drawdown_so_far': max_drawdown,
   515→   515→        })
   516→   516→
   517→   517→    final_price = df.iloc[-1]['price_thb']
   518→   518→    final_value = btc * final_price + cash_reserve
   519→   519→    final_avg_cost = total_invested / btc if btc > 0 else 0
   520→   520→    roi_pct = ((final_value - total_invested) / total_invested * 100) if total_invested > 0 else 0
   521→   521→    net_profit = final_value - total_invested
   522→   522→
   523→   523→    results = {
   524→   524→        'strategy': strategy_name,
   525→   525→        'total_invested': total_invested,
   526→   526→        'total_btc': btc,
   527→   527→        'avg_cost_thb': final_avg_cost,
   528→   528→        'avg_cost_usd': final_avg_cost / USD_THB_RATE,
   529→   529→        'final_value': final_value,
   530→   530→        'cash_reserve': cash_reserve,
   531→   531→        'roi_pct': roi_pct,
   532→   532→        'net_profit': net_profit,
   533→   533→        'max_drawdown_pct': max_drawdown * 100,
   534→   534→    }
   535→   535→    return results, pd.DataFrame(daily_log)

   801→   801→    # Precompute: MVRV 30-day momentum, SMA 200
   802→   802→    mvrv_series = df_precomputed['mvrv']
   803→   803→    mvrv_mom_30 = (mvrv_series / mvrv_series.shift(30) - 1).values  # % change over 30 days
   804→   804→    sma_200 = df_precomputed['price_usd'].rolling(200, min_periods=50).mean().values
   805→   805→
   806→   806→    def strategy_func(state):
   807→   807→        row = state['row']
   808→   808→        idx = state['idx']
   809→   809→        price_usd = row['price_usd']
   810→   810→        mvrv = row['mvrv']
   811→   811→        sopr = row['sopr']
   812→   812→        cash = state['cash_reserve']
   813→   813→        cooldown = state['cooldown']
   814→   814→
   815→   815→        # =============================================
   816→   816→        # 1. BASE TIERS: Absolute MVRV (like C, but tuned)
   817→   817→        #    Key: don't over-buy at "fair value" MVRV 1.5-2.0
   818→   818→        # =============================================
   819→   819→        if mvrv < 0.8:
   820→   820→            multiplier = 5.0   # Deep capitulation (C maxes at 4.5x)
   821→   821→        elif mvrv < 1.0:
   822→   822→            multiplier = 3.5   # Undervalued (C: 3.0-4.5x depending on SOPR)
   823→   823→        elif mvrv < 1.5:
   824→   824→            multiplier = 2.0   # Cheap (C: 2.0-3.0x)
   825→   825→        elif mvrv < 2.0:
   826→   826→            multiplier = 1.0   # Fair value — standard DCA (matches C exactly)
   827→   827→        elif mvrv < 2.5:
   828→   828→            multiplier = 0.3   # Expensive — minimal (C: 0.5x, Beta is stricter)
   829→   829→        else:
   830→   830→            multiplier = 0.0   # Very expensive — pause
   831→   831→
   832→   832→        # =============================================
   833→   833→        # 2. SOPR BOOSTER (only at true capitulation MVRV < 1.0)
   834→   834→        # =============================================
   835→   835→        if mvrv < 1.0 and sopr < 0.95:
   836→   836→            multiplier = 5.5  # Beats C's 4.5x
   837→   837→
   838→   838→        # =============================================
   839→   839→        # 3. MVRV 30-DAY MOMENTUM (Beta's unique edge)
   840→   840→        #    If MVRV is falling fast, the market is rapidly revaluing
   841→   841→        #    downward — a buying opportunity even at moderate levels.
   842→   842→        # =============================================
   843→   843→        mom = mvrv_mom_30[idx] if idx < len(mvrv_mom_30) and not np.isnan(mvrv_mom_30[idx]) else 0
   844→   844→        if mom < -0.20:        # MVRV dropped > 20% in 30 days
   845→   845→            multiplier *= 2.0
   846→   846→        elif mom < -0.10:    # MVRV dropped > 10% in 30 days
   847→   847→            multiplier *= 1.5
   848→   848→        # Cap at 6.0x (don't go insane)
   849→   849→        multiplier = min(multiplier, 6.0)
   850→   850→
   851→   851→        buy_amount = BASE_BUDGET_THB * multiplier
   852→   852→
   853→   853→        # =============================================
   854→   854→        # 4. SMART RESERVE (small, aggressively deployed)
   855→   855→        #    When expensive: route base budget to reserve
   856→   856→        #    When cheap: deploy 15% of reserve per day (3x faster than E)
   857→   857→        # =============================================
   858→   858→        to_reserve = 0.0
   859→   859→        if mvrv >= 2.5:
   860→   860→            to_reserve = BASE_BUDGET_THB  # Full base budget to reserve
   861→   861→
   862→   862→        if mvrv < 1.2 and cash > 0:
   863→   863→            injection = min(cash * 0.15, 1500)  # 15% per day, cap 1500 THB
   864→   864→            buy_amount += injection
   865→   865→            to_reserve -= injection
   866→   866→            to_reserve = max(to_reserve, 0.0)
   867→   867→
   868→   868→        # =============================================
   869→   869→        # 5. REGIME-AWARE SELLING
   870→   870→        #    Only sell in confirmed bull (price > SMA200)
   871→   871→        #    MVRV > 3.0 + is_bull → sell 8%, cooldown 45 days
   872→   872→        # =============================================
   873→   873→        sell_pct = 0.0
   874→   874→        new_cooldown = cooldown
   875→   875→        if cooldown == 0 and state['btc'] > 0:
   876→   876→            s200 = sma_200[idx] if idx < len(sma_200) else price_usd
   877→   877→            is_bull = price_usd > s200 if not np.isnan(s200) else True
   878→   878→            if is_bull and mvrv > 3.0:
   879→   879→                sell_pct = 8.0
   880→   880→                new_cooldown = 45
   881→   881→
   882→   882→        return {
   883→   883→            'buy_thb': buy_amount, 'sell_btc_pct': sell_pct,
   884→   884→            'to_reserve': to_reserve, 'new_cooldown': new_cooldown,
   885→   885→        }
   886→   886→
   887→   887→    return strategy_func
