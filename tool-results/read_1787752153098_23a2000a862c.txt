   106→   106→// ─── Mini Chart Component (SVG sparkline) ───
   107→   107→
   108→   108→function Sparkline({
   109→   109→  data,
   110→   110→  color = "#f59e0b",
   111→   111→  height = 40,
   112→   112→}: {
   113→   113→  data: number[];
   114→   114→  color?: string;
   115→   115→  height?: number;
   116→   116→}) {
   117→   117→  if (data.length < 2) return null;
   118→   118→  const min = Math.min(...data);
   119→   119→  const max = Math.max(...data);
   120→   120→  const range = max - min || 1;
   121→   121→  const w = 100;
   122→   122→  const points = data
   123→   123→    .map((v, i) => {
   124→   124→      const x = (i / (data.length - 1)) * w;
   125→   125→      const y = height - ((v - min) / range) * (height - 4) - 2;
   126→   126→      return `${x},${y}`;
   127→   127→    })
   128→   128→    .join(" ");
   129→   129→  return (
   130→   130→    <svg
   131→   131→      viewBox={`0 0 ${w} ${height}`}
   132→   132→      className="w-full"
   133→   133→      style={{ height }}
   134→   134→      preserveAspectRatio="none"
   135→   135→    >
   136→   136→      <polyline
   137→   137→        points={points}
   138→   138→        fill="none"
   139→   139→        stroke={color}
   140→   140→        strokeWidth="1.5"
   141→   141→        strokeLinejoin="round"
   142→   142→      />
   143→   143→    </svg>
   144→   144→  );
   145→   145→}
   146→   146→
   147→   147→// ─── Format helpers ───
   148→   148→
   149→   149→function fmt(n: number, decimals = 2): string {
   150→   150→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(decimals)}M`;
   151→   151→  if (n >= 1_000) return `${(n / 1_000).toFixed(decimals)}K`;
   152→   152→  return n.toFixed(decimals);
   153→   153→}
   154→   154→
   155→   155→function fmtCurrency(n: number, currency: string): string {
   156→   156→  return `${n.toLocaleString(undefined, { maximumFractionDigits: 0 })} ${currency}`;
   157→   157→}
   158→   158→
   159→   159→function ChangeBadge({ value }: { value: number | null }) {
   160→   160→  if (value === null) return <span className="text-muted-foreground text-xs">N/A</span>;
   161→   161→  const isUp = value >= 0;
   162→   162→  return (
   163→   163→    <span
   164→   164→      className={`inline-flex items-center gap-0.5 text-sm font-semibold ${isUp ? "text-emerald-500" : "text-red-500"}`}
   165→   165→    >
   166→   166→      {isUp ? <ArrowUpRight className="w-3.5 h-3.5" /> : <ArrowDownRight className="w-3.5 h-3.5" />}
   167→   167→      {Math.abs(value).toFixed(2)}%
   168→   168→    </span>
   169→   169→  );
   170→   170→}
