   663→function renderSparkline() {
   664→  const data = state.sparkline;
   665→  if (data.length < 2) return;
   666→
   667→  const svg = document.getElementById('sparkline');
   668→  const w = 600, h = 60;
   669→  const min = Math.min(...data);
   670→  const max = Math.max(...data);
   671→  const range = max - min || 1;
   672→
   673→  const points = data.map((v, i) => {
   674→    const x = (i / (data.length - 1)) * w;
   675→    const y = h - ((v - min) / range) * (h - 4) - 2;
   676→    return `${x},${y}`;
   677→  }).join(' ');
   678→
   679→  const areaPoints = `0,${h} ${points} ${w},${h}`;
   680→  const isUp = data[data.length-1] >= data[0];
   681→  const color = isUp ? '#22c55e' : '#ef4444';
   682→
   683→  svg.setAttribute('viewBox', `0 0 ${w} ${h}`);
   684→  svg.innerHTML = `
   685→    <defs>
   686→      <linearGradient id="sparkGrad" x1="0" y1="0" x2="0" y2="1">
   687→        <stop offset="0%" stop-color="${color}" stop-opacity="0.3"/>
   688→        <stop offset="100%" stop-color="${color}" stop-opacity="0"/>
   689→      </linearGradient>
   690→    </defs>
   691→    <polygon points="${areaPoints}" fill="url(#sparkGrad)"/>
   692→    <polyline points="${points}" fill="none" stroke="${color}" stroke-width="1.5" stroke-linejoin="round"/>
   693→    <circle cx="${(data.length-1)/(data.length-1)*w}" cy="${h - ((data[data.length-1]-min)/range)*(h-4)-2}" r="3" fill="${color}"/>
   694→  `;
   695→}

   871→  // MVRV = price / realized_price, so realized_price = price / MVRV
   872→  const rp = state.onchain.mvrv > 0 ? usd / state.onchain.mvrv : 0;
   873→  // LTH-RP approximation (typically 1.15x realized price)
   874→  const lthRp = rp * 1.15;
   875→
   876→  if (rp > 0) {
   877→    levels.push({ label: 'Realized Price (derived)', value: rp, dist: ((usd - rp) / rp) * 100 });
   878→  }
   879→  if (lthRp > 0) {
   880→    levels.push({ label: 'LTH Realized Price (est.)', value: lthRp, dist: ((usd - lthRp) / lthRp) * 100 });
   881→  }
   882→  levels.push({ label: 'ATH', value: CONFIG.ATH_USD, dist: ((usd - CONFIG.ATH_USD) / CONFIG.ATH_USD) * 100 });
   883→
   884→  // Add from bot state if available
   885→  if (state.bot && state.bot.last_indicators) {
   886→    const bi = state.bot.last_indicators;
   887→    if (bi.realized_price > 0) {
   888→      levels[0] = { label: 'Realized Price', value: bi.realized_price / 35, dist: ((usd - bi.realized_price/35) / (bi.realized_price/35)) * 100 };
   889→    }
   890→    if (bi.lth_realized_price > 0) {
   891→      levels[1] = { label: 'LTH Realized Price', value: bi.lth_realized_price / 35, dist: ((usd - bi.lth_realized_price/35) / (bi.lth_realized_price/35)) * 100 };
   892→    }
   893→  }
   894→
   895→  let html = '<div style="display:grid;grid-template-columns:repeat(auto-fit,minmax(200px,1fr));gap:0">';
   896→  levels.forEach(l => {
   897→    const dColor = l.dist >= 0 ? 'var(--green)' : 'var(--red)';
   898→    html += `<div class="level-row">`;
   899→    html += `  <span class="level-label">${l.label}</span>`;
   900→    html += `  <div class="level-right">`;
   901→    html += `    <span class="level-value">${fmtUSD(l.value)}</span>`;
   902→    html += `    <span class="level-dist" style="color:${dColor}">${l.dist >= 0 ? '+' : ''}${l.dist.toFixed(1)}%</span>`;
   903→    html += `  </div>`;
   904→    html += `</div>`;
   905→  });
   906→  html += '</div>';
   907→  html += `<div class="level-note">`;
   908→  html += `  Below Realized Price = historically profitable accumulation zone. `;
   909→  html += `  Above LTH-RP = long-term holders in profit (sell pressure zone). `;
   910→  html += `  Levels derived from MVRV: RP ≈ Price / MVRV.`;
   911→  html += `</div>`;
   912→
   913→  document.getElementById('levels-content').innerHTML = html;
   914→}
