    89→def _sopr_proxy(price, sma14, sma30):
    90→    """Estimate SOPR from price vs short-term moving average.
    91→
    92→    SOPR = price / avg_cost_basis_of_STH.
    93→    SMA14 approximates short-term holder average buy price.
    94→    price > SMA14 -> recent buyers in profit -> SOPR > 1
    95→    """
    96→    if sma14 > 0 and not math.isnan(sma14):
    97→        return price / sma14
    98→    if sma30 > 0 and not math.isnan(sma30):
    99→        return price / sma30
   100→    return 1.0

   103→def _lth_rp_proxy(realized_price_val, price, mvrv_val, in_bear=False):
   104→    """Estimate LTH Realized Price from Realized Price.
   105→
   106→    B23: Dynamic multiplier based on market regime.
   107→    LTH holders have higher cost basis than overall realized price.
   108→    """
   109→    if not math.isnan(realized_price_val) and realized_price_val > 0:
   110→        multiplier = LTH_RP_BEAR_MULTIPLIER if in_bear else (LTH_RP_HIGH_MVRV_MULTIPLIER if mvrv_val > HIGH_MVRV_THRESHOLD else LTH_RP_DEFAULT_MULTIPLIER)
   111→        return realized_price_val * multiplier
   112→    # Derive from MVRV if no realized price
   113→    if mvrv_val > 0 and not math.isnan(mvrv_val) and price > 0:
   114→        est_rp = price / mvrv_val
   115→        multiplier = LTH_RP_BEAR_MULTIPLIER if in_bear else (LTH_RP_HIGH_MVRV_MULTIPLIER if mvrv_val > HIGH_MVRV_THRESHOLD else LTH_RP_DEFAULT_MULTIPLIER)
   116→        return est_rp * multiplier
   117→    return float('nan')

   120→def _resolve_onchain_metrics(price, closes, today,
   121→                              log_prefix='[BOT]',
   122→                              allow_web_fallback=True,
   123→                              notify_on_fail=True):
   124→    """Resolve all technical + on-chain indicators with unified fallback chain.
   125→
   126→    B22: Replaces duplicated code in run_daily(), refresh_dashboard(),
   127→    run_demo(), and idempotency-skip path.
   128→
   129→    Steps:
   130→        1. Compute technical indicators from price closes
   131→        2. Resolve MVRV (BG cache -> embedded -> web)
   132→        3. Call BG batch for on-chain metrics
   133→        4. Apply proxy fallbacks for SOPR/LTH-RP
   134→
   135→    Args:
   136→        price: current BTC price
   137→        closes: list of daily close prices
   138→        today: date object (Thai TZ)
   139→        log_prefix: prefix for log messages (e.g. '[BOT]', '[REFRESH]', '[DEMO]')
   140→        allow_web_fallback: try CoinMetrics/ahasignals when embedded MVRV is missing
   141→        notify_on_fail: send Telegram when all MVRV sources fail
   142→
   143→    Returns:
   144→        dict with all resolved metrics. Key differences from callers:
   145→        - run_daily uses allow_web=True, notify_on_fail=True
   146→        - refresh uses allow_web=False, notify_on_fail=False
   147→        - demo uses allow_web=True, notify_on_fail=False
   148→    """
   149→    # ── 1. Technical indicators ──
   150→    sma_200 = ind.sma(closes, 200)
   151→    sma_365 = ind.sma(closes, 365)
   152→    rsi_val = ind.rsi(closes, 14)
   153→    macd_line, macd_sig, macd_h = ind.macd(closes)
   154→    macd_hist_series = ind.compute_all_macd_hist(closes)
   155→    rsi_series = ind.compute_all_rsi(closes, 14)
   156→    macd_bear = ind.macd_cross_bear(macd_hist_series)
   157→    macd_declining = ind.macd_hist_declining(macd_hist_series, 4)
   158→    rsi_div = ind.rsi_divergence(closes, rsi_series, 40)
   159→    ath = max(closes) if closes else 0
   160→    sma14 = ind.sma(closes, 14)
   161→    sma30 = ind.sma(closes, 30)
   162→
   163→    print(f'{log_prefix} SMA200={sma_200:,.2f} RSI={rsi_val:.1f} MACD_H={macd_h:.4f}')
   164→    print(f'{log_prefix} MACD_bear={macd_bear} MACD_declining={macd_declining} RSI_div={rsi_div}')
   165→
   166→    in_bear = not math.isnan(sma_200) and price < sma_200
   167→
   168→    # ── 2. MVRV — try BG cache first, then embedded, then web ──
   169→    mvrv_val = float('nan')
   170→    mvrv_z = float('nan')
   171→    mvrv_z_source = 'N/A'
   172→    mvrv_source = 'N/A'
   173→
   174→    try:
   175→        from . import bg_metrics
   176→        bg_mvrv = bg_metrics.get_cached_value('mvrv', today)
   177→        if not math.isnan(bg_mvrv):
   178→            mvrv_val = bg_mvrv
   179→            if mvrv_val is not None and mvrv_val <= 0:
   180→                mvrv_val = float('nan')
   181→            mvrv_source = 'BG-cache'
   182→            print(f'{log_prefix} MVRV from BG cache: {mvrv_val:.4f}')
   183→        bg_z = bg_metrics.get_cached_value('mvrv_zscore', today)
   184→        if not math.isnan(bg_z):
   185→            mvrv_z = bg_z
   186→            mvrv_z_source = 'BG-cache'
   187→    except ImportError:
   188→        pass
   189→
   190→    if math.isnan(mvrv_val):
   191→        mvrv_val = strategy.get_mvrv_for_date(today)
   192→        if mvrv_val is not None and mvrv_val <= 0:
   193→            mvrv_val = float('nan')
   194→        if not math.isnan(mvrv_val):
   195→            mvrv_source = 'embedded'
   196→            if allow_web_fallback:
   197→                from datetime import timedelta as td
   198→                if today - strategy._MVRV_HISTORY_MAX > td(days=1):
   199→                    print(f'{log_prefix} MVRV embedded stale (ends {strategy._MVRV_HISTORY_MAX}), '
   200→                          f'updating in background...')
   201→                    try:
   202→                        from . import mvrv_fetcher
   203→                        ok, msg = mvrv_fetcher.try_update_mvrv()
   204→                        print(f'{log_prefix} MVRV update: {msg}')
   205→                    except Exception as e:
   206→                        print(f'{log_prefix} MVRV background update failed: {e}')
   207→        elif allow_web_fallback:
   208→            print(f'{log_prefix} No embedded MVRV for {today}, trying web fallback...')
   209→            from . import mvrv_fetcher
   210→            web_mvrv, web_date, web_source = mvrv_fetcher.fetch_mvrv_from_web()
   211→            if web_mvrv is not None:
   212→                mvrv_val = web_mvrv
   213→                if mvrv_val is not None and mvrv_val <= 0:
   214→                    mvrv_val = float('nan')
   215→                mvrv_source = web_source
   216→                print(f'{log_prefix} Web MVRV: {mvrv_val:.4f}')
   217→            else:
   218→                print(f'{log_prefix} WARNING: All MVRV sources failed: {web_source}')
   219→                if notify_on_fail:
   220→                    notifier.send_telegram(
   221→                        f'Phoenix v5.1 WARNING: No MVRV data for {today}. '
   222→                        'All sources failed. Skipping trade.'
   223→                    )
   224→                return {'_mvrv_all_failed': True}
   225→        else:
   226→            mvrv_source = 'N/A'
   227→
   228→    # DI-6: Default to NaN (not 0) when MVRV unavailable.
   229→    # 0 falsely implies 'MVRV at minimum' to dashboard/strategy.
   230→    mvrv_pct = strategy.compute_mvrv_percentile(today, mvrv_val) if not math.isnan(mvrv_val) else float('nan')
   231→    if math.isnan(mvrv_z):
   232→        mvrv_z = strategy.compute_mvrv_zscore(today, mvrv_val) if not math.isnan(mvrv_val) else float('nan')
   233→        mvrv_z_source = 'embedded-365d'
   234→    nupl = 1.0 - 1.0 / mvrv_val if mvrv_val > 0 and not math.isnan(mvrv_val) else 0
   235→    realized_price = price / mvrv_val if mvrv_val > 0 and not math.isnan(mvrv_val) else float('nan')
   236→
   237→    # ── 3. BG batch fetch + on-chain metric resolution ──
   238→    sopr = float('nan')
   239→    lth_rp = float('nan')
   240→    sopr_source = 'N/A'
   241→    lth_source = 'N/A'
   242→    rp_source = 'mvrv-derived' if not math.isnan(realized_price) else 'N/A'
   243→
   244→    try:
   245→        from . import bg_metrics
   246→        bg = bg_metrics.get_all_metrics_today(target_date=today)
   247→
   248→        # MVRV upgrade from BG batch
   249→        if not math.isnan(bg.get('mvrv', float('nan'))):
   250→            bg_mvrv_val = bg['mvrv']
   251→            if bg_mvrv_val > 0 and mvrv_source != 'BG':
   252→                mvrv_val = bg_mvrv_val
   253→                mvrv_source = 'BG'
   254→                nupl = 1.0 - 1.0 / mvrv_val
   255→                realized_price = price / mvrv_val
   256→                rp_source = 'BG'
   257→                print(f'{log_prefix} MVRV upgraded from BG: {mvrv_val:.4f}')
   258→
   259→        # MVRV Z-Score from BG
   260→        if not math.isnan(bg.get('mvrv_zscore', float('nan'))):
   261→            mvrv_z = bg['mvrv_zscore']
   262→            mvrv_z_source = 'BG'
   263→            print(f'{log_prefix} MVRV Z-Score from BG: {mvrv_z:.3f}')
   264→
   265→        # SOPR
   266→        if not math.isnan(bg.get('sth_sopr', float('nan'))):
   267→            sopr = bg['sth_sopr']
   268→            sopr_source = 'BG'
   269→        else:
   270→            sopr = _sopr_proxy(price, sma14, sma30)
   271→            sopr_source = 'proxy-sma14'
   272→
   273→        # Realized Price
   274→        if not math.isnan(bg.get('realized_price', float('nan'))):
   275→            realized_price = bg['realized_price']
   276→            rp_source = 'BG'
   277→
   278→        # LTH Realized Price
   279→        if not math.isnan(bg.get('lth_realized_price', float('nan'))):
   280→            lth_rp = bg['lth_realized_price']
   281→            lth_source = 'BG'
   282→        else:
   283→            lth_rp = _lth_rp_proxy(realized_price, price, mvrv_val, in_bear=in_bear)
   284→            lth_source = 'proxy-rp*dynamic' if not math.isnan(lth_rp) else 'N/A'
   285→
   286→        # B20: Proxy accuracy logging (when BG has actual SOPR)
   287→        _proxy_sopr = _sopr_proxy(price, sma14, sma30)
   288→        if not math.isnan(sopr) and not math.isnan(_proxy_sopr) and sopr_source == 'BG':
   289→            proxy_err = abs(sopr - _proxy_sopr) / max(abs(sopr), SOPR_ACCURACY_FLOOR) * 100
   290→            print(f'{log_prefix} SOPR proxy accuracy: actual={sopr:.4f} proxy={_proxy_sopr:.4f} err={proxy_err:.1f}%')
   291→
   292→    except Exception as e:
   293→        print(f'{log_prefix} BG metrics failed: {e}. Using all-proxy mode')
   294→        sopr = _sopr_proxy(price, sma14, sma30)
   295→        lth_rp = _lth_rp_proxy(realized_price, price, mvrv_val, in_bear=in_bear)
   296→        sopr_source = 'proxy-sma14'
   297→        lth_source = 'proxy-rp*dynamic'
   298→
   299→    print(f'{log_prefix} STH-SOPR={sopr:.4f} ({sopr_source}) '
   300→          f'LTH-RP={lth_rp:,.2f} ({lth_source}) '
   301→          f'RP={realized_price:,.2f} ({rp_source})')
   302→    print(f'{log_prefix} MVRV={mvrv_val:.3f} ({mvrv_source}) Pct={mvrv_pct:.3f} Z={mvrv_z:.2f} ({mvrv_z_source}) NUPL={nupl:.3f}')
   303→
   304→    return {
   305→        'sma_200': sma_200, 'sma_365': sma_365,
   306→        'rsi': rsi_val, 'macd_line': macd_line, 'macd_sig': macd_sig, 'macd_h': macd_h,
   307→        'macd_hist_series': macd_hist_series, 'rsi_series': rsi_series,
   308→        'macd_bear': macd_bear, 'macd_declining': macd_declining, 'rsi_div': rsi_div,
   309→        'ath': ath, 'sma14': sma14, 'sma30': sma30, 'in_bear': in_bear,
   310→        'mvrv': mvrv_val, 'mvrv_source': mvrv_source,
   311→        'mvrv_pct': mvrv_pct, 'mvrv_z': mvrv_z, 'mvrv_z_source': mvrv_z_source,
   312→        'nupl': nupl, 'realized_price': realized_price, 'rp_source': rp_source,
   313→        'sopr': sopr, 'sopr_source': sopr_source,
   314→        'lth_realized_price': lth_rp, 'lth_source': lth_source,
   315→    }

   318→def _get_cash_balance(exchange) -> float:
   319→    '''Get available cash in exchange currency.'''
   320→    try:
   321→        if hasattr(exchange, 'get_usdt_balance'):
   322→            return exchange.get_usdt_balance()
   323→        # Prefer robust get_balances() if available (Bitkub)
   324→        if hasattr(exchange, 'get_balances'):
   325→            balances = exchange.get_balances()
   326→        elif hasattr(exchange, 'get_balance'):
   327→            balances = exchange.get_balance()
   328→        else:
   329→            return 0.0
   330→        if isinstance(balances, dict):
   331→            return balances.get(exchange.currency, 0.0)
   332→    except Exception as e:
   333→        print(f'[BOT] WARNING: Could not fetch cash balance: {e}')
   334→    return 0.0

   358→def _build_indicators_snapshot(metrics: dict, overrides: dict = None) -> dict:
   359→    """Build standardized indicators dict for state snapshot.
   360→
   361→    Used by run_daily(), refresh_dashboard(), run_demo(), and
   362→    idempotency-skip path. Eliminates ~80 lines of duplication.
   363→
   364→    Args:
   365→        metrics: dict from _resolve_onchain_metrics()
   366→        overrides: optional dict of extra fields to merge
   367→    """
   368→    snap = {
   369→        'price': round(metrics['price'], 2),
   370→        'mvrv': round(metrics['mvrv'], 3) if not math.isnan(metrics['mvrv']) else None,
   371→        'mvrv_source': metrics['mvrv_source'],
   372→        'mvrv_pct': round(metrics['mvrv_pct'], 3) if not math.isnan(metrics.get('mvrv_pct', float('nan'))) else None,
   373→        'mvrv_z': round(metrics['mvrv_z'], 2) if not math.isnan(metrics.get('mvrv_z', float('nan'))) else None,
   374→        'mvrv_z_source': metrics['mvrv_z_source'],
   375→        'rsi': round(metrics['rsi'], 1),
   376→        'macd_h': round(metrics['macd_h'], 4),
   377→        'nupl': round(metrics['nupl'], 3),
   378→        'sopr': round(metrics['sopr'], 3) if not math.isnan(metrics['sopr']) else None,
   379→        'sopr_source': metrics['sopr_source'],
   380→        'sma_200': round(metrics['sma_200'], 2) if not math.isnan(metrics['sma_200']) else None,
   381→        'sma_365': round(metrics['sma_365'], 2) if not math.isnan(metrics['sma_365']) else None,
   382→        'macd_bear': metrics['macd_bear'],
   383→        'macd_declining': metrics['macd_declining'],
   384→        'rsi_divergence': metrics['rsi_div'],
   385→        'ath': round(metrics['ath'], 2),
   386→        'sell_score': metrics.get('sell_score', 0),
   387→        'path_taken': metrics.get('path_taken', 'none'),
   388→        'in_bear': metrics.get('in_bear', False),
   389→        'cooldown': metrics.get('cooldown', 0),
   390→    }
   391→    # Optional fields (not always present in metrics)
   392→    for key in ('realized_price', 'lth_realized_price', 'lth_source', 'rp_source'):
   393→        val = metrics.get(key)
   394→        if val is not None:
   395→            if isinstance(val, float) and math.isnan(val):
   396→                snap[key] = None
   397→            elif isinstance(val, float):
   398→                snap[key] = round(val, 2)
   399→            else:
   400→                snap[key] = val
   401→    if overrides:
   402→        snap.update(overrides)
   403→    return snap

   454→def run_daily(exchange, bot_state: dict, dry_run: bool = False,
   455→              trade_log_path: str = 'trade_log.json',
   456→              kill_switch_path: str = 'kill_switch.json',
   457→              force: bool = False) -> dict:
   458→    '''Main daily run. Returns updated state.
   459→
   460→    Steps:
   461→        0. Idempotency guard (bypassed with --force)
   462→        -1. Kill switch check (L1 + L2)
   463→        1.  Fetch price & history
   464→        2.  Compute indicators
   465→        3.  Get MVRV
   466→        4.  Get balances (virtual in dry-run, real in live)
   467→        5.  Convert budget
   468→        6.  Decrement cooldown
   469→        7.  Run strategy
   470→        8.  Execute trades (simulated in dry-run, real in live)
   471→        9.  Update state + trade log
   472→        10. Snapshot indicators for dashboard
   473→        11. Send notification
   474→    '''
   475→    currency = exchange.currency
   476→    today = _thai_today()
   477→
   478→    # ── 0a. Print effective config (for debugging budget/multiplier issues) ──
   479→    print(f'[BOT] Config: DAILY_BUDGET_THB={config.DAILY_BUDGET_THB} MAX_BUY_THB={config.MAX_BUY_THB} '
   480→          f'MAX_DCA_BUYS_PER_DAY={config.MAX_DCA_BUYS_PER_DAY}')
   481→
   482→    # ── 0. Idempotency guard: skip if already ran today (unless --force) ──
   483→    # H1: Uses Thai timezone so daily guard aligns with THB calendar day
   484→    # (cron at 03:00/03:20/03:40 UTC = 10:00/10:20/10:40 THB)
   485→    # Each cron slot also has 3x internal retry with 60s backoff.
   486→    # Backup slots are safety nets — if 1st run succeeds,
   487→    # last_run_date is set and later runs skip via this guard.
   488→    # If 1st run fails (exception/timeout), last_run_date is NOT updated,
   489→    # so the next cron slot proceeds normally.
   490→    if not force and bot_state.get('last_run_date') == today.isoformat():
   491→        print(f'[BOT] Already ran today ({today} THB). Skipping TRADE but refreshing dashboard data.')
   492→        print(f'[BOT] Use --force to override (e.g. for testing).')
   493→        # Still fetch indicators + balances for dashboard (no trading)
   494→        # B22: Use shared _resolve_onchain_metrics instead of duplicated inline code
   495→        try:
   496→            refresh_price = exchange.get_price()
   497→            refresh_klines = _fetch_price_history_with_dates(exchange)
   498→            refresh_closes = [k['close'] for k in refresh_klines]
   499→            if len(refresh_closes) >= MIN_CLOSES_FOR_INDICATORS:
   500→                rm = _resolve_onchain_metrics(refresh_price, refresh_closes, today,
   501→                                                log_prefix='[BOT/idem]',
   502→                                                allow_web_fallback=False,
   503→                                                notify_on_fail=False)
   504→                if not rm.get('_mvrv_all_failed'):
   505→                    # Balances
   506→                    if dry_run:
   507→                        r_btc = bot_state.get('dry_run_btc', 0.0)
   508→                        r_cash = bot_state.get('dry_run_cash', config.DRY_RUN_INITIAL_CASH)
   509→                    else:
   510→                        r_btc = _get_btc_balance(exchange)
   511→                        r_cash = _get_cash_balance(exchange)
   512→                    r_portfolio = r_btc * refresh_price + r_cash
   513→
   514→                    rm_for_snap = {**rm, 'price': refresh_price,
   515→                                   'sell_score': bot_state.get('last_indicators', {}).get('sell_score', 0),
   516→                                   'path_taken': bot_state.get('last_indicators', {}).get('path_taken', 'none'),
   517→                                   'in_bear': rm['in_bear'],
   518→                                   'cooldown': bot_state.get('cooldown', 0)}
   519→                    bot_state['last_indicators'] = _build_indicators_snapshot(rm_for_snap)
   520→                    _update_state_metadata(bot_state, exchange, r_btc, r_cash, r_portfolio, refresh_price)
   521→                    # D2: Do NOT overwrite last_dry_run in refresh-only paths.
   522→                    if r_portfolio > bot_state.get('peak_value', 0):
   523→                        bot_state['peak_value'] = r_portfolio
   524→                    print(f'[BOT] Dashboard data refreshed (skipped trade). Portfolio: {r_portfolio:,.2f} {currency}')
   525→        except Exception as e:
   526→            print(f'[BOT] Dashboard refresh failed: {e}')
   527→        return bot_state
   528→
   529→    # ── 0b. Daily buy count guard (MAX_DCA_BUYS_PER_DAY) ──
   530→    # This is a SECONDARY guard on top of the idempotency guard above.
   531→    # It counts actual buys in trade_log.json for today (Thai date),
   532→    # preventing runaway buys from: manual workflow_dispatch with --force,
   533→    # stale state from failed git push, or concurrent process race conditions.
   534→    max_daily_buys = config.MAX_DCA_BUYS_PER_DAY
   535→    today_str = today.isoformat()
   536→    try:
   537→        existing_log = state_mod.load_trade_log(trade_log_path)
   538→        today_buy_count = sum(
   539→            1 for t in existing_log
   540→            if t.get('type') == 'buy' and t.get('date', '').startswith(today_str)
   541→        )
   542→    except Exception:
   543→        today_buy_count = 0
   544→
   545→    if not force and today_buy_count >= max_daily_buys:
   546→        print(f'[BOT] Daily buy limit reached: {today_buy_count}/{max_daily_buys} buys today ({today}). Skipping.')
   547→        notifier.send_telegram(
   548→            f'Phoenix v5.1: Daily buy limit BLOCKED {today_buy_count+1}th buy '
   549→            f'({today_buy_count}/{max_daily_buys} already done today).'
   550→        )
   551→        # Still update last_run_date to prevent re-running the full pipeline
   552→        bot_state['last_run_date'] = today_str
   553→        bot_state['run_count'] += 1
   554→        return bot_state
   555→
   556→    # ── 0c. Dry-run → Live transition reset (D3) ──
   557→    # When transitioning from dry-run to first live run, reset all
   558→    # counters and metrics that were contaminated by dry-run data.
   559→    # This prevents phantom invested/BTC/count values on the dashboard.
   560→    if not dry_run and bot_state.get('last_dry_run') is True:
   561→        print('[BOT] D3: Dry-run → Live transition detected. Resetting contaminated state.')
   562→        for _key in ('total_invested', 'adjusted_invested', 'total_sell_proceeds',
   563→                     'total_btc_bought', 'total_btc_sold', 'cumulative_fees',
   564→                     'peak_value', 'max_drawdown', 'sell_proceeds_reserve',
   565→                     'dry_run_sell_proceeds', 'total_reserve_injected'):
   566→            bot_state[_key] = 0.0 if isinstance(bot_state.get(_key, 0), (int, float)) else None
   567→        for _key in ('buy_count', 'sell_count'):
   568→            bot_state[_key] = 0
   569→        bot_state['last_trade_date'] = ''
   570→        bot_state['last_sell_date'] = ''
   571→        bot_state['realized_price'] = None
   572→        bot_state['lth_realized_price'] = None
   573→        bot_state['dry_run_btc'] = None
   574→        bot_state['dry_run_cash'] = None
   575→        # D3-ext: Also clear trade_log.json to remove contaminated dry-run entries.
   576→        # The dashboard's D1 filter (dry_run=False check) handles this at read time,
   577→        # but cleaning the source file is the correct approach.
   578→        try:
   579→            state_mod.clear_trade_log(trade_log_path)
   580→        except Exception as e:
   581→            print(f'[BOT] D3: WARNING — failed to clear trade log: {e}')
   582→        print('[BOT] D3: State reset complete. Starting fresh with live data.')
   583→
   584→    # ── -1. Kill Switch Check ──
   585→    is_alive, kill_reason = ks_mod.check_kill_switch(kill_switch_path)
   586→    if not is_alive:
   587→        print(f'[BOT] KILLED: {kill_reason}')
   588→        # Still fetch indicators for dashboard, but skip all trading
   589→        try:
   590→            price = exchange.get_price()
   591→            _snapshot_indicators(bot_state, price, currency, dry_run,
   592→                                  exchange, killed=True, kill_reason=kill_reason)
   593→        except Exception as e:
   594→            print(f'[BOT] Could not fetch indicators for dashboard: {e}')
   595→        notifier.send_telegram(
   596→            f'Phoenix v5.1 KILLED: {kill_reason}\n'
   597→            f'No trades executed. Dashboard still updated.'
   598→        )
   599→        # Do NOT set last_run_date — a killed run must not consume
   600→        # the daily idempotency slot so a re-run can still trade.
   601→        bot_state['run_count'] += 1
   602→        return bot_state
   603→
   604→    # ── 1. Fetch current price ──
   605→    print(f'[BOT] Fetching price from {exchange.__class__.__name__}...')
   606→    price = exchange.get_price()
   607→    print(f'[BOT] Current price: {price:,.2f} {currency}')
   608→
   609→    if price <= 0:
   610→        print(f'[BOT] ERROR: Invalid price from exchange: {price}. Skipping.')
   611→        notifier.send_telegram(
   612→            f'Phoenix v5.1 ERROR: Invalid price from exchange: {price}. Skipping.'
   613→        )
   614→        return bot_state
   615→
   616→    # ── 2. Fetch price history for indicators ──
   617→    print('[BOT] Fetching price history (500d)...')
   618→    klines = _fetch_price_history_with_dates(exchange)
   619→    closes = [k['close'] for k in klines]
   620→    print(f'[BOT] Got {len(closes)} daily closes')
   621→
   622→    if len(closes) < MIN_CLOSES_FOR_INDICATORS:
   623→        print('[BOT] ERROR: Not enough price history for indicators')
   624→        notifier.send_telegram(
   625→            f'Phoenix v5.1 ERROR: Only {len(closes)} days of price data. Skipping.'
   626→        )
   627→        return bot_state
   628→
   629→    # ── 3+4+4b. Resolve all indicators + on-chain metrics (B22: shared function) ──
   630→    print('[BOT] Computing indicators + resolving on-chain metrics...')
   631→    m = _resolve_onchain_metrics(price, closes, today,
   632→                                  log_prefix='[BOT]',
   633→                                  allow_web_fallback=True,
   634→                                  notify_on_fail=True)
   635→    if m.get('_mvrv_all_failed'):
   636→        return bot_state
   637→
   638→    # Unpack resolved metrics
   639→    sma_200 = m['sma_200']
   640→    sma_365 = m['sma_365']
   641→    rsi_val = m['rsi']
   642→    macd_h = m['macd_h']
   643→    macd_bear = m['macd_bear']
   644→    macd_declining = m['macd_declining']
   645→    rsi_div = m['rsi_div']
   646→    ath = m['ath']
   647→    mvrv_val = m['mvrv']
   648→    mvrv_source = m['mvrv_source']
   649→    mvrv_pct = m['mvrv_pct']
   650→    mvrv_z = m['mvrv_z']
   651→    mvrv_z_source = m['mvrv_z_source']
   652→    nupl = m['nupl']
   653→    realized_price = m['realized_price']
   654→    rp_source = m['rp_source']
   655→    sopr = m['sopr']
   656→    sopr_source = m['sopr_source']
   657→    lth_rp = m['lth_realized_price']
   658→    lth_source = m['lth_source']
   659→    in_bear = m['in_bear']
   660→
   661→    # ── 5. Get exchange balances ──
   662→    # ╔═══════════════════════════════════════════════════════════════╗
   663→    # ║  DRY RUN SAFETY: Virtual balances — NO real API balance calls ║
   664→    # ║  LIVE MODE: Real balances fetched from exchange API           ║
   665→    # ╚═══════════════════════════════════════════════════════════════╝
   666→    if dry_run:
   667→        # DRY RUN: Use virtual balances from state (no real API calls)
   668→        if bot_state.get('dry_run_cash') is None:
   669→            bot_state['dry_run_cash'] = config.DRY_RUN_INITIAL_CASH
   670→            bot_state['dry_run_btc'] = 0.0
   671→            print(f'[BOT] DRY RUN: Initialized virtual wallet')
   672→            print(f'[BOT]   Virtual cash: {config.DRY_RUN_INITIAL_CASH:,.2f} {currency}')
   673→        btc_balance = bot_state.get('dry_run_btc', 0.0)
   674→        cash_balance = bot_state.get('dry_run_cash', config.DRY_RUN_INITIAL_CASH)
   675→        print(f'[BOT] DRY RUN: Virtual BTC={btc_balance:.8f} Cash={cash_balance:,.2f} {currency}')
   676→    else:
   677→        # LIVE MODE: Fetch real balances from exchange
   678→        btc_balance = _get_btc_balance(exchange)
   679→        cash_balance = _get_cash_balance(exchange)
   680→        print(f'[BOT] LIVE MODE: Real BTC={btc_balance:.8f} Cash={cash_balance:,.2f} {currency}')
   681→
   682→    # ── 6. Convert budget to exchange currency ──
   683→    base_budget = config.get_daily_budget()
   684→    max_buy = config.get_max_buy()
   685→    print(f'[BOT] Budget: {base_budget:.2f} {currency}/run (max buy: {max_buy:.2f})')
   686→
   687→    # ── 6a. Monday DCA boost (CQ: shared helper) ──
   688→    # Research: Monday has highest next-day BTC returns (+0.38% avg, 6/7 sources).
   689→    # Apply to base_budget BEFORE strategy so max_buy cap inside strategy still works.
   690→    base_budget, monday_boost = _apply_monday_boost(base_budget)
   691→    if monday_boost:
   692→        print(f'[BOT] MONDAY BOOST: base_budget x{config.MONDAY_DCA_MULTIPLIER} = {base_budget:.2f} {currency}')
   693→
   694→    # ── 6b. DCA time window check ──
   695→    # Only buy during configured Thai time window (default: 10:00-11:00 THB).
   696→    # Indicators + dashboard are still computed outside the window.
   697→    in_dca_window = _in_dca_time_window()
   698→    if not in_dca_window:
   699→        thai_h = _thai_now().hour
   700→        print(f'[BOT] DCA TIME WINDOW: {thai_h}:xx THB is outside '
   701→              f'{config.DCA_TIME_WINDOW_START}:00-{config.DCA_TIME_WINDOW_END}:00. '
   702→              f'Skipping BUY (indicators + dashboard still updated).')
   703→
   704→    # ── 6c. Separate reserve from DCA cash ──
   705→    # In demo: sell_proceeds_reserve is tracked separately
   706→    # In live/dry-run: we track total_sell_proceeds in state as reserve proxy
   707→    # cash_reserve passed to strategy = ONLY profits from BTC sales
   708→    sell_proceeds_reserve = bot_state.get('total_sell_proceeds', 0.0) - bot_state.get('total_reserve_injected', 0.0)
   709→    sell_proceeds_reserve = max(sell_proceeds_reserve, 0.0)
   710→    # For dry-run, also consider virtual sell proceeds
   711→    if dry_run and bot_state.get('dry_run_sell_proceeds'):
   712→        sell_proceeds_reserve = bot_state['dry_run_sell_proceeds']
   713→    print(f'[BOT] Sell proceeds reserve: {sell_proceeds_reserve:,.2f} {currency}')
   714→
   715→    # ── 7. Decrement cooldown ──
   716→    if bot_state.get('cooldown', 0) > 0:
   717→        bot_state['cooldown'] -= 1
   718→        print(f'[BOT] Cooldown: {bot_state["cooldown"] + 1} -> {bot_state["cooldown"]}')
   719→
   720→    # ── 8. Run strategy ──
   721→    print('[BOT] Running Phoenix v5.1 strategy...')
   722→    decision = strategy.phoenix_v5_1_decision(
   723→        mvrv=mvrv_val, rsi=rsi_val, sopr=sopr, nupl=nupl,
   724→        price=price, sma_200=sma_200, sma_365=sma_365,
   725→        realized_price=realized_price, lth_realized_price=lth_rp,
   726→        mvrv_pct=mvrv_pct, mvrv_z=mvrv_z,
   727→        macd_cross_bear=macd_bear, macd_hist_declining=macd_declining,
   728→        rsi_divergence_flag=rsi_div, ath=ath,
   729→        btc_balance=btc_balance,
   730→        cash_reserve=sell_proceeds_reserve,  # Only BTC sale profits, not DCA cash
   731→        cooldown=bot_state['cooldown'],
   732→        base_budget=base_budget, max_buy=max_buy,
   733→        # Reserve deployment config (all in exchange currency)
   734→        reserve_floor=config.get_reserve_floor(),
   735→        max_reserve_injection=config.get_max_reserve_injection(),
   736→        max_reserve_boosted=config.get_max_reserve_boosted(),
   737→        reserve_boost_multiplier=config.RESERVE_BOOST_MULTIPLIER,
   738→        reserve_boost_price_ratio=config.RESERVE_BOOST_PRICE_RATIO,
   739→    )
   740→
   741→    print(f'[BOT] Decision: buy={decision["buy_amount"]:.2f} '
   742→          f'sell={decision["sell_amount"]:.2f} '
   743→          f'score={decision["sell_score"]} path={decision["path_taken"]}')
   744→
   745→    # ── 8b. DCA time window enforcement ──
   746→    # Zero out buy if outside the configured Thai time window.
   747→    # This is a safety feature that ALWAYS applies, even with --force.
   748→    # Rationale: buying outside the research-optimized window defeats the purpose.
   749→    # Sell decisions are NOT affected by time window.
   750→    if not in_dca_window and decision['buy_amount'] > 0:
   751→        print(f'[BOT] TIME WINDOW BLOCK: Zeroing buy {decision["buy_amount"]:.2f} (outside DCA window)')
   752→        decision['buy_amount'] = 0.0
   753→
   754→    # ── 9. Execute trades ──
   755→    buy_fee = 0.0
   756→    sell_fee = 0.0
   757→    buy_btc_got = 0.0
   758→    sell_btc_sold = 0.0
   759→    buy_cost_actual = 0.0
   760→    sell_proceeds_actual = 0.0
   761→    trade_attempted = False
   762→    trade_succeeded = False
   763→    # BUY
   764→    if decision['buy_amount'] > 0:
   765→        # Check minimum order size
   766→        min_buy = config.MIN_BUY_USDT if currency == 'USDT' else config.MIN_BUY_THB
   767→        if decision['buy_amount'] < min_buy:
   768→            print(f'[BOT] Buy amount {decision["buy_amount"]:.2f} below minimum {min_buy}. Skipping.')
   769→            decision['buy_amount'] = 0
   770→        elif cash_balance < decision['buy_amount']:
   771→            print(f'[BOT] Insufficient cash: have {cash_balance:,.2f}, need {decision["buy_amount"]:.2f}')
   772→            # Buy with what we have
   773→            if cash_balance >= min_buy:
   774→                decision['buy_amount'] = cash_balance
   775→            else:
   776→                decision['buy_amount'] = 0
   777→
   778→    # ═══════════════════════════════════════════════════════════════
   779→    # DRY RUN SAFETY GATE
   780→    # When dry_run=True, exchange.market_buy() is NEVER called.
   781→    # Trades are simulated locally with virtual balances only.
   782→    # ═══════════════════════════════════════════════════════════════
   783→    if decision['buy_amount'] > 0 and not dry_run:
   784→        trade_attempted = True
   785→        print(f'[BOT] LIVE BUY: {decision["buy_amount"]:.2f} {currency} of BTC...')
   786→        try:
   787→            result = exchange.market_buy(decision['buy_amount'])
   788→            buy_cost_actual = float(result.get('cummulative_quote_qty', result.get('cost', decision['buy_amount'])))
   789→            buy_btc_got = float(result.get('executed_qty', result.get('amount', 0)))
   790→            # C3: Fallback — if API returns 0, compute BTC from cost/price
   791→            if buy_btc_got <= 0 and buy_cost_actual > 0 and price > 0:
   792→                print(f'[BOT] C3 fallback: executed_qty=0, computing BTC = {buy_cost_actual:.2f} / {price:,.2f}')
   793→                buy_btc_got = buy_cost_actual / price
   794→            buy_fee = float(result.get('fee', buy_cost_actual * config.BUY_FEE_PCT))
   795→            bot_state['total_btc_bought'] += buy_btc_got
   796→            trade_succeeded = True
   797→            print(f'[BOT] Bought {buy_btc_got:.8f} BTC for {buy_cost_actual:.2f} {currency} (fee: {buy_fee:.2f})')
   798→            print(f'[BOT] BUY STATUS: SUCCESS')
   799→        except Exception as e:
   800→            print(f'[BOT] BUY ERROR: {e}')
   801→            print(f'[BOT] BUY STATUS: FAILED')
   802→            # C2: If timeout/connection error, order likely executed on exchange.
   803→            # Consume daily slot to prevent double-buy on retry.
   804→            # Also estimate the trade details so trade_log and state are updated.
   805→            _err_str = str(e).lower()
   806→            if 'timeout' in _err_str or 'connection' in _err_str or 'ssl' in _err_str:
   807→                print(f'[BOT] BUY TIMEOUT — attempting verification via balance check...')
   808→                trade_succeeded = True  # Consume daily slot to prevent double-buy
   809→                time.sleep(BALANCE_VERIFY_DELAY_S)
   810→                try:
   811→                    post_btc = _get_btc_balance(exchange)
   812→                    btc_diff = post_btc - btc_balance
   813→                    if btc_diff > BTC_BALANCE_EPSILON:  # BTC balance increased — order likely executed
   814→                        buy_btc_got = btc_diff
   815→                        buy_cost_actual = decision['buy_amount']  # best estimate of cost
   816→                        buy_fee = 0  # Unknown — be conservative
   817→                        bot_state['total_btc_bought'] += buy_btc_got
   818→                        print(f'[BOT] TIMEOUT VERIFIED: +{btc_diff:.8f} BTC detected via balance check')
   819→                    else:
   820→                        # BTC unchanged — order likely did NOT execute
   821→                        buy_btc_got = 0
   822→                        buy_cost_actual = 0
   823→                        buy_fee = 0
   824→                        decision['buy_amount'] = 0  # DI-1: prevent phantom invested/buy_count
   825→                        print(f'[BOT] TIMEOUT UNVERIFIED: BTC balance unchanged ({post_btc:.8f}). '
   826→                              f'Slot consumed to prevent double-buy. No trade recorded.')
   827→                except Exception as ve:
   828→                    buy_btc_got = 0
   829→                    buy_cost_actual = 0
   830→                    buy_fee = 0
   831→                    decision['buy_amount'] = 0  # DI-1: prevent phantom invested/buy_count
   832→                    print(f'[BOT] TIMEOUT VERIFICATION FAILED: {ve}. '
   833→                          f'Slot consumed to prevent double-buy. No trade recorded.')
   834→            else:
   835→                decision['buy_amount'] = 0
   836→    elif decision['buy_amount'] > 0 and dry_run:
   837→        if price <= 0:
   838→            print(f'[BOT] DRY RUN BUY SKIP: invalid price {price}')
   839→            decision['buy_amount'] = 0
   840→        else:
   841→            buy_btc_got = decision['buy_amount'] / price
   842→            buy_cost_actual = decision['buy_amount']
   843→            buy_fee = buy_cost_actual * config.BUY_FEE_PCT
   844→            bot_state['total_btc_bought'] += buy_btc_got
   845→            print(f'[BOT] DRY RUN BUY: {decision["buy_amount"]:.2f} {currency} → {buy_btc_got:.8f} BTC @ {price:,.2f} (fee: {buy_fee:.2f})')
   846→
   847→    # SELL
   848→    # ═══════════════════════════════════════════════════════════════
   849→    # DRY RUN SAFETY GATE (SELL)
   850→    # When dry_run=True, exchange.market_sell() is NEVER called.
   851→    # ═══════════════════════════════════════════════════════════════
   852→    if decision['sell_amount'] > 0 and not dry_run:
   853→        if price <= 0:
   854→            print(f'[BOT] LIVE SELL SKIP: invalid price {price}')
   855→            decision['sell_amount'] = 0
   856→        else:
   857→            btc_to_sell = decision['sell_amount'] / price
   858→            if btc_to_sell >= btc_balance * MAX_SELL_PCT_OF_BALANCE:
   859→                btc_to_sell = btc_balance * MAX_SELL_PCT_OF_BALANCE  # Never sell 100%
   860→            min_sell = MIN_SELL_AMOUNT
   861→            if btc_to_sell * price < min_sell:
   862→                print(f'[BOT] Sell amount {btc_to_sell * price:.2f} below minimum {min_sell}. Skipping.')
   863→                decision['sell_amount'] = 0
   864→            else:
   865→                trade_attempted = True
   866→                print(f'[BOT] LIVE SELL: {btc_to_sell:.8f} BTC (~{decision["sell_amount"]:.2f} {currency})...')
   867→                try:
   868→                    result = exchange.market_sell(btc_to_sell)
   869→                    sell_btc_sold = float(result.get('executed_qty', result.get('amount', 0)))
   870→                    sell_proceeds_actual = float(result.get('cummulative_quote_qty', result.get('cost', decision['sell_amount'])))
   871→                    sell_fee = float(result.get('fee', sell_proceeds_actual * config.SELL_FEE_PCT))
   872→                    bot_state['total_btc_sold'] += sell_btc_sold
   873→                    # Update cash reserve after sell
   874→                    cash_balance += sell_proceeds_actual
   875→                    trade_succeeded = True
   876→                    print(f'[BOT] Sold {sell_btc_sold:.8f} BTC for {sell_proceeds_actual:.2f} {currency} (fee: {sell_fee:.2f})')
   877→                except Exception as e:
   878→                    print(f'[BOT] SELL ERROR: {e}')
   879→                    decision['sell_amount'] = 0
   880→                    decision['new_cooldown'] = 0
   881→    elif decision['sell_amount'] > 0 and dry_run:
   882→        if price <= 0:
   883→            print(f'[BOT] DRY RUN SELL SKIP: invalid price {price}')
   884→            decision['sell_amount'] = 0
   885→        else:
   886→            sell_btc_sold = decision['sell_amount'] / price
   887→            sell_proceeds_actual = decision['sell_amount']
   888→            sell_fee = sell_proceeds_actual * config.SELL_FEE_PCT
   889→            bot_state['total_btc_sold'] += sell_btc_sold
   890→            print(f'[BOT] DRY RUN SELL: {sell_btc_sold:.8f} BTC → {sell_proceeds_actual:.2f} {currency} (fee: {sell_fee:.2f})')
   891→
   892→    # Record trades in trade log FIRST (before mutating state)
   893→    # This ensures if trade log write fails, state is not yet mutated,
   894→    # preventing data inconsistency between state.json and trade_log.json.
   895→    # H3: Compute actual fill prices from exchange response for accurate records
   896→    buy_fill_price = price
   897→    if buy_btc_got > 0 and buy_cost_actual > 0:
   898→        buy_fill_price = buy_cost_actual / buy_btc_got
   899→
   900→    sell_fill_price = price
   901→    if sell_btc_sold > 0 and sell_proceeds_actual > 0:
   902→        sell_fill_price = sell_proceeds_actual / sell_btc_sold
   903→
   904→    if decision['buy_amount'] > 0 and buy_btc_got > 0:
   905→        buy_extra = {'dry_run': dry_run,
   906→                     'reserve': round(decision.get('reserve_injection', 0), 2)}
   907→        if monday_boost:
   908→            buy_extra['monday_boost'] = config.MONDAY_DCA_MULTIPLIER
   909→        state_mod.append_trade_log(
   910→            trade_log_path, 'buy', buy_cost_actual, buy_btc_got,
   911→            buy_fill_price, buy_fee,
   912→            extra=buy_extra
   913→        )
   914→
   915→    if decision['sell_amount'] > 0 and sell_btc_sold > 0:
   916→        state_mod.append_trade_log(
   917→            trade_log_path, 'sell', sell_proceeds_actual, sell_btc_sold,
   918→            sell_fill_price, sell_fee,
   919→            extra={'dry_run': dry_run,
   920→                   'path': decision.get('path_taken', ''),
   921→                   'score': decision.get('sell_score', 0)}
   922→        )
   923→
   924→    # ── 10. Update state ──
   925→    bot_state = state_mod.update_state_after_run(
   926→        bot_state, decision, buy_fill_price, sell_fill_price, currency,
   927→        buy_fee=buy_fee, sell_fee=sell_fee,
   928→        btc_balance=btc_balance, cash_balance=cash_balance,
   929→        sell_proceeds_actual=sell_proceeds_actual,
   930→        actual_buy_cost=buy_cost_actual,
   931→    )
   932→
   933→    # If a trade was attempted but failed (amount zeroed), don't consume daily slot
   934→    if trade_attempted and not trade_succeeded:
   935→        bot_state.pop('last_run_date', None)
   936→        bot_state['run_count'] -= 1  # Revert the increment
   937→        print(f'[BOT] Trade failed — not consuming daily slot. Cron retry will try again.')
   938→
   939→    # Track portfolio value
   940→    if dry_run:
   941→        # DRY RUN: Use virtual balances (no exchange API calls)
   942→        current_btc = btc_balance + buy_btc_got - sell_btc_sold
   943→        current_cash = cash_balance - buy_cost_actual + sell_proceeds_actual - buy_fee - sell_fee
   944→        # Save updated virtual balances to state for next run
   945→        bot_state['dry_run_btc'] = round(current_btc, 8)
   946→        bot_state['dry_run_cash'] = round(current_cash, 2)
   947→        # Track sell proceeds for reserve
   948→        if sell_proceeds_actual > 0:
   949→            bot_state['dry_run_sell_proceeds'] = bot_state.get('dry_run_sell_proceeds', 0.0) + sell_proceeds_actual - sell_fee
   950→        print(f'[BOT] DRY RUN portfolio: BTC={current_btc:.8f} Cash={current_cash:,.2f} {currency}')
   951→    else:
   952→        # LIVE: Fetch latest real balances from exchange
   953→        current_btc = _get_btc_balance(exchange)
   954→        current_cash = _get_cash_balance(exchange)
   955→        # Track sell proceeds for reserve (sell_proceeds_actual is net-of-fee for Bitkub)
   956→        if sell_proceeds_actual > 0:
   957→            bot_state['sell_proceeds_reserve'] = bot_state.get('sell_proceeds_reserve', 0.0) + sell_proceeds_actual
   958→    portfolio = current_btc * price + current_cash
   959→
   960→    peak = bot_state.get('peak_value', 0.0)
   961→    if portfolio > peak:
   962→        bot_state['peak_value'] = portfolio
   963→        peak = portfolio
   964→    if peak > 0:
   965→        dd = (bot_state['peak_value'] - portfolio) / bot_state['peak_value']
   966→        if dd > bot_state['max_drawdown']:
   967→            bot_state['max_drawdown'] = dd
   968→
   969→    # ── 11. Snapshot indicators for dashboard ──
   970→    m_for_snap = {**m, 'price': price,
   971→                   'sell_score': decision.get('sell_score', 0),
   972→                   'path_taken': decision.get('path_taken', 'none'),
   973→                   'in_bear': decision.get('in_bear', False),
   974→                   'cooldown': decision.get('new_cooldown', 0)}
   975→    bot_state['last_indicators'] = _build_indicators_snapshot(m_for_snap)
   976→    bot_state['last_decision'] = _build_decision_metadata(decision, monday_boost, in_dca_window)
   977→    _update_state_metadata(bot_state, exchange, current_btc, current_cash, portfolio, price)
   978→    bot_state['last_dry_run'] = dry_run
   979→
   980→    # ── 11b. B18: Append indicator history for retrospective analysis ──
   981→    try:
   982→        ih_path = os.path.join(os.path.dirname(trade_log_path), 'indicator_history.json')
   983→        state_mod.append_indicator_history(
   984→            ih_path,
   985→            bot_state.get('last_indicators', {}),
   986→            bot_state.get('last_decision', {}),
   987→        )
   988→        print(f'[BOT] Indicator history appended')
   989→    except Exception as e:
   990→        print(f'[BOT] Indicator history append failed (non-critical): {e}')
   991→
   992→    # ── 12. Low balance warning ──
   993→    daily_budget = config.get_daily_budget()
   994→    if current_cash > 0 and daily_budget > 0:
   995→        days_remaining = current_cash / daily_budget
   996→        if days_remaining <= config.LOW_BALANCE_DAYS:
   997→            warning_msg = (
   998→                f'\n⚠ LOW BALANCE WARNING\n'
   999→                f'Cash: {current_cash:,.2f} {currency}\n'
  1000→                f'Daily budget: {daily_budget:,.2f} {currency}\n'
  1001→                f'Runs remaining: ~{days_remaining:.1f}\n'
  1002→                f'Consider adding funds or reducing DCA budget.'
  1003→            )
  1004→            print(f'[BOT] {warning_msg}')
  1005→            notifier.send_telegram(warning_msg)
  1006→
  1007→    # ── 13. Send notification ──
  1008→    # DI-4: Pass actual fill amounts for accurate Telegram display
  1009→    msg = notifier.format_report(
  1010→        decision, price, mvrv_val, current_btc, current_cash,
  1011→        currency, is_dry_run=dry_run,
  1012→        monday_boost=config.MONDAY_DCA_MULTIPLIER if monday_boost else 1.0,
  1013→        actual_buy=buy_cost_actual,
  1014→        actual_sell=sell_proceeds_actual,
  1015→    )
  1016→    if notifier.send_telegram(msg):
  1017→        print('[BOT] Telegram notification sent')
  1018→    else:
  1019→        print('[BOT] Telegram not configured or failed')
  1020→
  1021→    print(f'[BOT] Done. Portfolio: {portfolio:,.2f} {currency}')
  1022→    return bot_state
