   266→def test_risk5_proxy_detection():
   267→    """RISK 5: Proxy Detection Accuracy"""
   268→    print("\n" + "="*70)
   269→    print("  RISK 5: Proxy Detection Accuracy")
   270→    print("="*70)
   271→    master = build_master_dataframe(years=5)
   272→
   273→    prices = master['price_usd'].values
   274→    sma365 = master['sma_365'].values
   275→    mvrv = master['mvrv'].values
   276→    is_real = master['mvrv_is_real'].values
   277→    dates = master['date'].values
   278→
   279→    fp = fn = tp = tn = 0
   280→    for i in range(len(master)):
   281→        pv = prices[i] / sma365[i] if sma365[i] > 0 else 999
   282→        detected = abs(mvrv[i] - pv) < 0.15 if not np.isnan(pv) else False
   283→        actually_proxy = not is_real[i]
   284→        if detected and actually_proxy: tp += 1
   285→        elif detected and not actually_proxy: fp += 1
   286→        elif not detected and actually_proxy: fn += 1
   287→        else: tn += 1
   288→
   289→    total = len(master)
   290→    accuracy = (tp + tn) / total * 100
   291→    print(f"  Real MVRV days: {is_real.sum()}")
   292→    print(f"  Proxy MVRV days: {total - is_real.sum()}")
   293→    print(f"  True Positives: {tp} | False Positives: {fp}")
   294→    print(f"  True Negatives: {tn} | False Negatives: {fn}")
   295→    print(f"  Detection accuracy: {accuracy:.1f}%")
   296→
   297→    if fp > 0:
   298→        print(f"\n  [MODERATE] {fp} real MVRV days incorrectly flagged as proxy")
   299→        fp_idx = np.where((np.array([abs(mvrv[i] - (prices[i]/sma365[i] if sma365[i] > 0 else 999)) < 0.15 for i in range(len(master))])) & (is_real))[0]
   300→        for idx in fp_idx[:5]:
   301→            pv = prices[idx] / sma365[idx] if sma365[idx] > 0 else 999
   302→            print(f"    FP: {dates[idx]} | MVRV={mvrv[idx]:.3f} | Proxy={pv:.3f} | Diff={abs(mvrv[idx]-pv):.3f}")
   303→    else:
   304→        print(f"  [LOW] Proxy detection works well")
   305→
   306→    return {'accuracy': accuracy, 'fp': fp, 'fn': fn}
