   308→   308→   308→def compute_technical_indicators(df):
   309→   309→   309→    """Add EMA 20, RSI 14, and SMA 365 to the DataFrame."""
   310→   310→   310→    prices = df['price_usd']
   311→   311→   311→    df['ema_20'] = prices.ewm(span=20, adjust=False).mean()
   312→   312→   312→    delta = prices.diff()
   313→   313→   313→    gain = delta.where(delta > 0, 0.0).ewm(span=14, adjust=False).mean()
   314→   314→   314→    loss = (-delta.where(delta < 0, 0.0)).ewm(span=14, adjust=False).mean()
   315→   315→   315→    rs = gain / loss.replace(0, np.nan)
   316→   316→   316→    df['rsi_14'] = 100 - (100 / (1 + rs))
   317→   317→   317→    df['rsi_14'] = df['rsi_14'].fillna(50)
   318→   318→   318→    df['sma_365'] = prices.rolling(365, min_periods=1).mean()
   319→   319→   319→    return df
