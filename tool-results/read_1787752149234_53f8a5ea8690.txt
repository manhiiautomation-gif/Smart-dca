   106→// ─── Mini Chart Component (SVG sparkline) ───
   107→
   108→function Sparkline({
   109→  data,
   110→  color = "#f59e0b",
   111→  height = 40,
   112→}: {
   113→  data: number[];
   114→  color?: string;
   115→  height?: number;
   116→}) {
   117→  if (data.length < 2) return null;
   118→  const min = Math.min(...data);
   119→  const max = Math.max(...data);
   120→  const range = max - min || 1;
   121→  const w = 100;
   122→  const points = data
   123→    .map((v, i) => {
   124→      const x = (i / (data.length - 1)) * w;
   125→      const y = height - ((v - min) / range) * (height - 4) - 2;
   126→      return `${x},${y}`;
   127→    })
   128→    .join(" ");
   129→  return (
   130→    <svg
   131→      viewBox={`0 0 ${w} ${height}`}
   132→      className="w-full"
   133→      style={{ height }}
   134→      preserveAspectRatio="none"
   135→    >
   136→      <polyline
   137→        points={points}
   138→        fill="none"
   139→        stroke={color}
   140→        strokeWidth="1.5"
   141→        strokeLinejoin="round"
   142→      />
   143→    </svg>
   144→  );
   145→}
   146→
   147→// ─── Format helpers ───
   148→
   149→function fmt(n: number, decimals = 2): string {
   150→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(decimals)}M`;
   151→  if (n >= 1_000) return `${(n / 1_000).toFixed(decimals)}K`;
   152→  return n.toFixed(decimals);
   153→}
   154→
   155→function fmtCurrency(n: number, currency: string): string {
   156→  return `${n.toLocaleString(undefined, { maximumFractionDigits: 0 })} ${currency}`;
   157→}
   158→
   159→function ChangeBadge({ value }: { value: number | null }) {
   160→  if (value === null) return <span className="text-muted-foreground text-xs">N/A</span>;
   161→  const isUp = value >= 0;
   162→  return (
   163→    <span
   164→      className={`inline-flex items-center gap-0.5 text-sm font-semibold ${isUp ? "text-emerald-500" : "text-red-500"}`}
   165→    >
   166→      {isUp ? <ArrowUpRight className="w-3.5 h-3.5" /> : <ArrowDownRight className="w-3.5 h-3.5" />}
   167→      {Math.abs(value).toFixed(2)}%
   168→    </span>
   169→  );
   170→}

   655→
   656→function LevelRow({
   657→  label,
   658→  value,
   659→  distance,
   660→  isDark,
   661→}: {
   662→  label: string;
   663→  value: number;
   664→  distance: number | null;
   665→  isDark: boolean;
   666→}) {
   667→  if (value <= 0) return null;
   668→  return (
   669→    <div className="flex items-center justify-between">
   670→      <span className={`text-xs ${isDark ? "text-zinc-400" : "text-zinc-500"}`}>
   671→        {label}
   672→      </span>
   673→      <div className="flex items-center gap-2">
   674→        <span className="text-xs font-mono">
   675→          {value.toLocaleString(undefined, { maximumFractionDigits: 0 })}
   676→        </span>
   677→        {distance !== null && (
   678→          <span
   679→            className={`text-xs font-semibold ${distance >= 0 ? "text-emerald-400" : "text-red-400"}`}
   680→          >
   681→            {distance >= 0 ? "+" : ""}
   682→            {distance.toFixed(1)}%
   683→          </span>
   684→        )}
   685→      </div>
   686→    </div>
   687→  );
   688→}
