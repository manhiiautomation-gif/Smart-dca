   122→def extract_events(daily_df, prev_sell_total=None):
   123→    """Extract sell events and reserve buy-back events from daily log."""
   124→    events = {'sell': [], 'buyback': []}
   125→    prev_st = 0.0
   126→    for i, row in daily_df.iterrows():
   127→        cur_st = row['sell_event_thb']
   128→        if prev_st is not None and cur_st > prev_st:
   129→            sell_amt = cur_st - prev_st
   130→            events['sell'].append({
   131→                'date': row['date'],
   132→                'price_thb': row['price_thb'],
   133→                'price_usd': row['price_usd'],
   134→                'amount_thb': sell_amt,
   135→                'portfolio': row['portfolio_value'],
   136→            })
   137→        if row['reserve_event_thb'] > 0:
   138→            events['buyback'].append({
   139→                'date': row['date'],
   140→                'price_thb': row['price_thb'],
   141→                'price_usd': row['price_usd'],
   142→                'amount_thb': row['reserve_event_thb'],
   143→                'portfolio': row['portfolio_value'],
   144→            })
   145→        prev_st = cur_st
   146→    return events

   211→                         where=path_b, alpha=0.20, color=C_MVRV_ZONE, zorder=1,

   215→    ax2.axhline(y=2.5, color=C_SELL, linewidth=0.8, linestyle='--', alpha=0.5)

   217→
   218→    # Combined legend
   219→    lines1, labels1 = ax.get_legend_handles_labels()
   220→    lines2, labels2 = ax2.get_legend_handles_labels()
   221→    all_lines = lines1 + lines2
   222→    all_labels = labels1 + labels2
   223→    if all_lines:
   224→        ax.legend(all_lines, all_labels, loc='upper left', fontsize=8, framealpha=0.95)
   225→

   227→    ax.xaxis.set_major_locator(mdates.MonthLocator(interval=max(3, years_label // 3)))
   228→    plt.setp(ax.xaxis.get_majorticklabels(), rotation=30, ha='right')

   381→            else:
   382→                best_ci = min(vals, key=vals.get)
   383→                table.get_celld()[(row_idx, best_ci)].set_facecolor('#DCFCE7')
   384→                table.get_celld()[(row_idx, best_ci)].set_text_props(fontweight='bold', fontsize=8.5)

   386→    return table
