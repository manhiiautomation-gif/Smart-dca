   206→def fetch_bgeometrics_metric(metric_name, token='7NqNRwWhyc'):
   207→    """
   208→    Fetch REAL on-chain metric from BGeometrics API (FALLBACK only).
   209→    Used for SOPR which CoinMetrics doesn't offer for free.
   210→    """
   211→    cache_key = f'bgeometrics_{metric_name}'
   212→    cached = _load_cache(cache_key)
   213→    if cached is not None:
   214→        return cached
   215→
   216→    url = f"https://api.bgeometrics.com/v1/{metric_name}?token={token}"
   217→    for attempt in range(3):
   218→        try:
   219→            if attempt > 0:
   220→                wait = 10 * attempt
   221→                print(f"[DATA] BGeometrics {metric_name}: retry in {wait}s (attempt {attempt+1}/3)")
   222→                time.sleep(wait)
   223→            resp = requests.get(url, headers={'User-Agent': 'Mozilla/5.0'}, timeout=15)
   224→            if resp.status_code == 429:
   225→                print(f"[DATA] BGeometrics {metric_name}: HTTP 429 rate-limit (attempt {attempt+1}/3)")
   226→                continue
   227→            if resp.status_code != 200:
   228→                return None
   229→            data = resp.json()
   230→            if not isinstance(data, list) or len(data) == 0:
   231→                return None
   232→            records = []
   233→            for item in data:
   234→                date_str = item.get('d')
   235→                metric_val = item.get(metric_name)
   236→                if date_str and metric_val is not None:
   237→                    dt = pd.to_datetime(date_str).date()
   238→                    records.append({'date': dt, metric_name: float(metric_val)})
   239→            if not records:
   240→                return None
   241→            df = pd.DataFrame(records).drop_duplicates(subset='date').sort_values('date').reset_index(drop=True)
   242→            print(f"[DATA] BGeometrics {metric_name}: {len(df)} records "
   243→                  f"({df['date'].iloc[0]} to {df['date'].iloc[-1]})")
   244→            _save_cache(cache_key, df)
   245→            return df
   246→        except Exception as e:
   247→            if attempt == 2:
   248→                return None
   249→    return None

   308→def compute_technical_indicators(df):
   309→    """Add EMA 20, RSI 14, SMA 365, MACD(12,26,9), and SMA 200 to the DataFrame."""
   310→    prices = df['price_usd']
   311→    df['ema_20'] = prices.ewm(span=20, adjust=False).mean()
   312→    delta = prices.diff()
   313→    gain = delta.where(delta > 0, 0.0).ewm(span=14, adjust=False).mean()
   314→    loss = (-delta.where(delta < 0, 0.0)).ewm(span=14, adjust=False).mean()
   315→    rs = gain / loss.replace(0, np.nan)
   316→    df['rsi_14'] = 100 - (100 / (1 + rs))
   317→    df['rsi_14'] = df['rsi_14'].fillna(50)
   318→    df['sma_365'] = prices.rolling(365, min_periods=1).mean()
   319→    df['sma_200'] = prices.rolling(200, min_periods=50).mean()
   320→    # MACD (12, 26, 9)
   321→    ema12 = prices.ewm(span=12, adjust=False).mean()
   322→    ema26 = prices.ewm(span=26, adjust=False).mean()
   323→    df['macd_line'] = ema12 - ema26
   324→    df['macd_signal'] = df['macd_line'].ewm(span=9, adjust=False).mean()
   325→    df['macd_hist'] = df['macd_line'] - df['macd_signal']
   326→    # Realized Price = Price / MVRV (exact derivation)
   327→    df['realized_price'] = df['price_usd'] / df['mvrv'].clip(lower=0.01)
   328→    return df

   458→def apply_buy_fee(thb_amount):
   459→    return thb_amount * (1 - BUY_FEE_PCT)

   461→def apply_sell_fee(thb_amount):
   462→    return thb_amount * (1 - SELL_FEE_PCT)

   464→def backtest_strategy(df, strategy_func, strategy_name):
   465→    """
   466→    Generic backtest runner. Calls strategy_func for each day.
   467→    """
   468→    btc = 0.0
   469→    cash_reserve = 0.0
   470→    total_invested = 0.0
   471→    cooldown = 0
   472→    peak_value = 0.0
   473→    max_drawdown = 0.0
   474→    daily_log = []
   475→
   476→    for idx, row in df.iterrows():
   477→        price_thb = row['price_thb']
   478→        if cooldown > 0:
   479→            cooldown -= 1
   480→
   481→        state = {
   482→            'btc': btc, 'cash_reserve': cash_reserve,
   483→            'total_invested': total_invested, 'cooldown': cooldown,
   484→            'row': row, 'idx': idx
   485→        }
   486→        action = strategy_func(state)
   487→
   488→        buy_thb = action.get('buy_thb', 0)
   489→        sell_btc_pct = action.get('sell_btc_pct', 0)
   490→        sell_thb = action.get('sell_thb', 0)
   491→        to_reserve = action.get('to_reserve', 0)
   492→        sell_score = action.get('sell_score', 0)
   493→
   494→        actual_buy = apply_buy_fee(buy_thb)
   495→        btc_bought = actual_buy / price_thb if price_thb > 0 else 0
   496→        btc += btc_bought
   497→        total_invested += buy_thb
   498→
   499→        if sell_btc_pct > 0 and btc > 0:
   500→            btc_to_sell = btc * (sell_btc_pct / 100.0)
   501→            sell_proceeds = apply_sell_fee(btc_to_sell * price_thb)
   502→            btc -= btc_to_sell
   503→            cash_reserve += sell_proceeds
   504→            cooldown = action.get('new_cooldown', cooldown)
   505→
   506→        # THB-based selling (Style Beta v3)
   507→        if sell_thb > 0 and btc > 0 and price_thb > 0:
   508→            btc_to_sell = sell_thb / price_thb
   509→            if btc_to_sell > btc:
   510→                btc_to_sell = btc
   511→            sell_proceeds = apply_sell_fee(btc_to_sell * price_thb)
   512→            btc -= btc_to_sell
   513→            cash_reserve += sell_proceeds
   514→            cooldown = action.get('new_cooldown', cooldown)
   515→
   516→        cash_reserve += to_reserve
   517→        cash_reserve = max(cash_reserve, 0.0)
   518→
   519→        portfolio_value = btc * price_thb + cash_reserve
   520→        avg_cost = total_invested / btc if btc > 0 else 0
   521→
   522→        if portfolio_value > peak_value:
   523→            peak_value = portfolio_value
   524→        if peak_value > 0:
   525→            drawdown = (peak_value - portfolio_value) / peak_value
   526→            if drawdown > max_drawdown:
   527→                max_drawdown = drawdown
   528→
   529→        daily_log.append({
   530→            'date': row['date'], 'price_thb': price_thb,
   531→            'btc': btc, 'cash_reserve': cash_reserve,
   532→            'total_invested': total_invested,
   533→            'portfolio_value': portfolio_value,
   534→            'avg_cost': avg_cost,
   535→            'max_drawdown_so_far': max_drawdown,
   536→        })
   537→
   538→    final_price = df.iloc[-1]['price_thb']
   539→    final_value = btc * final_price + cash_reserve
   540→    final_avg_cost = total_invested / btc if btc > 0 else 0
   541→    roi_pct = ((final_value - total_invested) / total_invested * 100) if total_invested > 0 else 0
   542→    net_profit = final_value - total_invested
   543→
   544→    results = {
   545→        'strategy': strategy_name,
   546→        'total_invested': total_invested,
   547→        'total_btc': btc,
   548→        'avg_cost_thb': final_avg_cost,
   549→        'avg_cost_usd': final_avg_cost / USD_THB_RATE,
   550→        'final_value': final_value,
   551→        'cash_reserve': cash_reserve,
   552→        'roi_pct': roi_pct,
   553→        'net_profit': net_profit,
   554→        'max_drawdown_pct': max_drawdown * 100,
   555→    }
   556→    return results, pd.DataFrame(daily_log)
